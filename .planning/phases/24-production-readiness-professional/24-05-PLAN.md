```tsx
import { FileText, AlertTriangle, CheckCircle, Clock } from "lucide-react";

export interface OperationalDocument {
  id: string;
  document_type: string;
  document_number: string | null;
  issued_at: string | null;
  expiry_date: string | null;
  issuing_authority: string | null;
  verification_status: "pending" | "verified" | "rejected";
  notes: string | null;
  created_at: string;
}

interface Props {
  documents: OperationalDocument[];
  uploadHref: string;  // URL for the upload button (plan 24-05 stubs this to a form page)
}

const VERIFICATION_BADGE: Record<
  string,
  { label: string; tone: string; Icon: typeof CheckCircle }
> = {
  pending:  { label: "Pendente",   tone: "orange", Icon: Clock },
  verified: { label: "Verificado", tone: "green",  Icon: CheckCircle },
  rejected: { label: "Rejeitado",  tone: "red",    Icon: AlertTriangle },
};

const DOCUMENT_TYPE_LABELS: Record<string, string> = {
  license:             "Carta de Condução",
  passport:            "Passaporte",
  bi:                  "Bilhete de Identidade",
  insurance:           "Seguro",
  inspection:          "Inspecção",
  contract_copy:       "Cópia de Contrato",
  commercial_register: "Registo Comercial",
  nuit_certificate:    "Certificado NUIT",
  other:               "Outro",
};

function daysUntil(dateStr: string | null): number | null {
  if (!dateStr) return null;
  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / 86_400_000);
}

export function OperationalDocumentsList({ documents, uploadHref }: Props) {
  return (
    <div>
      <div className="flex justify-between items-center mb-4">
        <h3 className="text-base font-semibold">Documentos Operacionais</h3>
        <a href={uploadHref} className="btn-primary btn-sm flex items-center gap-2">
          <FileText size={14} /> Upload Documento
        </a>
      </div>

      {documents.length === 0 ? (
        <div className="text-center py-8 text-muted text-sm">
          Sem documentos registados
        </div>
      ) : (
        <div className="table-wrap">
          <table className="table">
            <thead>
              <tr>
                <th>Tipo</th>
                <th>Número</th>
                <th>Emissão</th>
                <th>Validade</th>
                <th>Entidade</th>
                <th>Estado</th>
              </tr>
            </thead>
            <tbody>
              {documents.map((doc) => {
                const days = daysUntil(doc.expiry_date);
                const isExpiring = days !== null && days >= 0 && days <= 30;
                const isExpired = days !== null && days < 0;
                const badge = VERIFICATION_BADGE[doc.verification_status] ??
                              VERIFICATION_BADGE.pending;
                const BadgeIcon = badge.Icon;

                return (
                  <tr key={doc.id}>
                    <td className="font-medium">
                      {DOCUMENT_TYPE_LABELS[doc.document_type] ?? doc.document_type}
                    </td>
                    <td className="font-mono text-sm">{doc.document_number ?? "—"}</td>
                    <td className="font-mono text-sm">{doc.issued_at ?? "—"}</td>
                    <td>
                      {doc.expiry_date ? (
                        <span className={`font-mono text-sm ${
                          isExpired ? "text-error font-medium"
                          : isExpiring ? "text-warning font-medium"
                          : "text-foreground"
                        }`}>
                          {doc.expiry_date}
                          {isExpired && " (vencido)"}
                          {isExpiring && !isExpired && ` (${days}d)`}
                        </span>
                      ) : "—"}
                    </td>
                    <td className="text-sm text-muted">{doc.issuing_authority ?? "—"}</td>
                    <td>
                      <span className={`badge badge-${badge.tone} flex items-center gap-1 w-fit`}>
                        <span className="dot" />
                        <BadgeIcon size={10} />
                        {badge.label}
                      </span>
                    </td>
                  </tr>
                );
              })}
            </tbody>
          </table>
        </div>
      )}
    </div>
  );
}
```

```tsx
"use client";

import { useState } from "react";
import { Upload } from "lucide-react";

interface DocumentUploadModalProps {
  subjectType: "driver" | "vehicle" | "third_party" | "client" | "contract";
  subjectId: string;
  onSuccess?: () => void;
}

const DOCUMENT_TYPES = [
  { value: "license",             label: "Carta de Condução" },
  { value: "passport",            label: "Passaporte" },
  { value: "bi",                  label: "Bilhete de Identidade" },
  { value: "insurance",           label: "Seguro" },
  { value: "inspection",          label: "Inspecção" },
  { value: "commercial_register", label: "Registo Comercial" },
  { value: "nuit_certificate",    label: "Certificado NUIT" },
  { value: "contract_copy",       label: "Cópia de Contrato" },
  { value: "other",               label: "Outro" },
];

export function DocumentUploadModal({
  subjectType, subjectId, onSuccess
}: DocumentUploadModalProps) {
  const [open, setOpen] = useState(false);
  const [form, setForm] = useState({
    document_type: "",
    document_number: "",
    issued_at: "",
    expiry_date: "",
    issuing_authority: "",
    notes: "",
  });
  const [submitting, setSubmitting] = useState(false);
  const [error, setError] = useState<string | null>(null);

  async function handleSubmit(e: React.FormEvent) {
    e.preventDefault();
    setSubmitting(true);
    setError(null);
    try {
      const res = await fetch("/api/operational-documents", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ ...form, subject_type: subjectType, subject_id: subjectId }),
      });
      if (!res.ok) {
        const body = await res.json();
        setError(body.detail ?? "Erro ao guardar documento");
        return;
      }
      setOpen(false);
      setForm({
        document_type: "", document_number: "", issued_at: "",
        expiry_date: "", issuing_authority: "", notes: "",
      });
      onSuccess?.();
    } catch {
      setError("Erro de rede — tente novamente");
    } finally {
      setSubmitting(false);
    }
  }

  if (!open) {
    return (
      <button onClick={() => setOpen(true)}
              className="btn-primary btn-sm flex items-center gap-2">
        <Upload size={14} /> Upload Documento
      </button>
    );
  }

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/50">
      <div className="bg-surface rounded-xl shadow-2xl w-full max-w-lg p-6">
        <h2 className="text-lg font-semibold mb-4">Registar Documento</h2>
        {error && <p className="text-error text-sm mb-3">{error}</p>}

        <form onSubmit={handleSubmit} className="space-y-3">
          <div>
            <label className="label-xs">Tipo de Documento *</label>
            <select required className="input-sm w-full"
                    value={form.document_type}
                    onChange={(e) => setForm((f) => ({
                      ...f, document_type: e.target.value
                    }))}>
              <option value="">Seleccionar tipo</option>
              {DOCUMENT_TYPES.map((t) => (
                <option key={t.value} value={t.value}>{t.label}</option>
              ))}
            </select>
          </div>

          <div className="grid grid-cols-2 gap-3">
            <div>
              <label className="label-xs">Número do documento</label>
              <input type="text" className="input-sm w-full"
                     value={form.document_number}
                     onChange={(e) => setForm((f) => ({
                       ...f, document_number: e.target.value
                     }))} />
            </div>
            <div>
              <label className="label-xs">Entidade emissora</label>
              <input type="text" className="input-sm w-full"
                     value={form.issuing_authority}
                     onChange={(e) => setForm((f) => ({
                       ...f, issuing_authority: e.target.value
                     }))} />
            </div>
          </div>

          <div className="grid grid-cols-2 gap-3">
            <div>
              <label className="label-xs">Data de emissão</label>
              <input type="date" className="input-sm w-full"
                     value={form.issued_at}
                     onChange={(e) => setForm((f) => ({
                       ...f, issued_at: e.target.value
                     }))} />
            </div>
            <div>
              <label className="label-xs">Data de validade</label>
              <input type="date" className="input-sm w-full"
                     value={form.expiry_date}
                     onChange={(e) => setForm((f) => ({
                       ...f, expiry_date: e.target.value
                     }))} />
            </div>
          </div>

          <div>
            <label className="label-xs">Notas</label>
            <textarea className="input-sm w-full" rows={2}
                      value={form.notes}
                      onChange={(e) => setForm((f) => ({
                        ...f, notes: e.target.value
                      }))} />
          </div>

          <div className="flex justify-end gap-3 pt-2">
            <button type="button" onClick={() => setOpen(false)} className="btn-ghost">
              Cancelar
            </button>
            <button type="submit" disabled={submitting} className="btn-primary">
              {submitting ? "A guardar..." : "Guardar"}
            </button>
          </div>
        </form>
      </div>
    </div>
  );
}
```

**Action:** CREATE

**Details:**
```typescript
import { NextRequest, NextResponse } from "next/server";
import { requireSession } from "@/app/lib/auth";

const API_BASE = process.env.NEXT_PUBLIC_API_URL ?? "http://localhost:8000";

export async function POST(request: NextRequest) {
  const session = await requireSession();
  const body = await request.json();
  const res = await fetch(`${API_BASE}/api/v1/third-party/documents`, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      Authorization: `Bearer ${session.accessToken}`,
      "X-Tenant-Id": session.tenantId,
    },
    body: JSON.stringify(body),
  });
  const data = await res.json();
  return NextResponse.json(data, { status: res.status });
}
```

```tsx
import { requireSession } from "@/app/lib/auth";

import { OperationalDocumentsList } from "@/app/components/OperationalDocumentsList";

import { notFound } from "next/navigation";
import Link from "next/link";

export default async function VehicleDetailPage({ params }: { params: { id: string } }) {
  await requireSession();

  const [vehicle, assignments, documents] = await Promise.all([
    apiFetch<Record<string, unknown>>(`/api/v1/vehicles/${params.id}`).catch(() => null),
    apiFetch<{ items?: unknown[]; assignments?: unknown[] }>(
      `/api/v1/third-party/assignments?vehicle_id=${params.id}`
    ).catch(() => ({ items: [] })),
    apiFetch<OperationalDocument[]>(
      `/api/v1/third-party/documents?subject_type=vehicle&subject_id=${params.id}`
    ).catch(() => []),
  ]);

  if (!vehicle) notFound();

  const assignmentList = (assignments as { items?: unknown[] }).items ??
                         (assignments as { assignments?: unknown[] }).assignments ?? [];

  return (
    <SidebarLayout active="viaturas">
      <div className="page-header mb-6">
        <div>
          <h1 className="text-2xl font-bold font-mono">{String(vehicle.plate ?? "")}</h1>
          <p className="text-muted text-sm mt-1">
            {String(vehicle.make ?? "")} {String(vehicle.model ?? "")} ·{" "}
            {String(vehicle.year ?? "")}
          </p>
        </div>
        <div className="flex gap-2">
          <StatusBadge
            tone={vehicle.status === "active" ? "green"
                   : vehicle.status === "maintenance" ? "orange" : "red"}
            label={vehicle.status === "active" ? "Activa"
                    : vehicle.status === "maintenance" ? "Manutenção" : "Inactiva"}
          />
          <Link href={`/viaturas/${params.id}/historico`} className="btn-secondary btn-sm">
            Histórico
          </Link>
        </div>
      </div>

      {/* Motoristas Atribuídos */}
      <section className="panel mb-6">
        <div className="flex justify-between items-center mb-4">
          <h2 className="text-lg font-semibold">Motoristas Atribuídos</h2>
          {/* Assignment action — links to create assignment form (Phase 24 scope: display only) */}
          <a href={`/viaturas/${params.id}/atribuir`} className="btn-secondary btn-sm">
            Atribuir Motorista
          </a>
        </div>
        {assignmentList.length === 0 ? (
          <p className="text-muted text-sm">Sem motoristas atribuídos</p>
        ) : (
          <div className="table-wrap">
            <table className="table">
              <thead>
                <tr>
                  <th>Motorista</th>
                  <th>Início</th>
                  <th>Fim</th>
                  <th>Estado</th>
                </tr>
              </thead>
              <tbody>
                {(assignmentList as Record<string, unknown>[]).map((a, i) => (
                  <tr key={String(a.id ?? i)}>
                    <td className="font-medium">{String(a.driver_name ?? a.driver_id ?? "—")}</td>
                    <td className="font-mono text-sm">{String(a.start_date ?? "—")}</td>
                    <td className="font-mono text-sm">{String(a.end_date ?? "Actual")}</td>
                    <td>
                      <StatusBadge
                        tone={!a.end_date ? "green" : "neutral"}
                        label={!a.end_date ? "Activa" : "Encerrada"}
                      />
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>

      {/* Documents section */}

        <OperationalDocumentsList
          documents={documents}
```
