## Goal

Add the `task_check_document_expiry` ARQ task to `worker.py` that scans all tenants for expiring operational documents and creates alerts using the existing `create_alert` service — idempotent, no new tables needed.

## Tasks

### Task 1 — ARQ task + cron registration

**File:** `backend/app/worker.py`

The worker uses an admin (BYPASSRLS) session that operates cross-tenant. Follow the exact pattern of `task_mark_overdue_billing_documents`:
- Import inside the function body (avoids circular imports at module load)
- Use `ctx["db_factory"]()` to open the session
- Log with structlog

Add the following function to `worker.py` BEFORE the `WorkerSettings` class:

```python
# ── TP-10: Document expiry alert cron ─────────────────────────────────────


async def task_check_document_expiry(ctx: dict) -> str:
    """TP-10: Daily cron — generate alerts for operational_documents expiring in 7 or 30 days.

    Uses BYPASSRLS admin session (cross-tenant scan).
    Alert deduplication: create_alert() is idempotent on request_reference.
    request_reference format: 'doc_expiry:{doc_id}:{expiry_date.isoformat()}'
    — includes doc.id to prevent collisions between documents with same expiry date.

    Runs daily at 06:00 Africa/Maputo = 04:00 UTC.
    """
    from datetime import date

    import structlog
    from sqlalchemy import and_, select

    from app.modules.alerts.schemas import AlertCreate
    from app.modules.alerts.service import create_alert
    from app.modules.third_party.models import OperationalDocument

    logger = structlog.get_logger("worker")
    today = date.today()
    alert_count = 0

    async with ctx["db_factory"]() as db:
        # Query all tenants' documents expiring within 30 days
        # Admin session bypasses RLS — no tenant_id filter needed here
        result = await db.execute(
            select(OperationalDocument).where(
                and_(
                    OperationalDocument.expiry_date.isnot(None),
                    OperationalDocument.expiry_date >= today,
                    OperationalDocument.expiry_date <= today.__class__.fromordinal(
                        today.toordinal() + 30
                    ),
                )
            )
        )
        docs = result.scalars().all()

        for doc in docs:
            days_remaining = (doc.expiry_date - today).days

            if days_remaining <= 7:
                priority = "critical"
                alert_type = "document_expiring_soon"
            else:
                priority = "high"
                alert_type = "document_expiring_soon"

            # Deterministic request_reference: includes doc.id to prevent cross-document collisions
            request_reference = f"doc_expiry:{doc.id}:{doc.expiry_date.isoformat()}"

            payload = AlertCreate(
                request_reference=request_reference,
                alert_type=alert_type,
                priority=priority,
                entity_type=doc.subject_type,   # 'driver' | 'vehicle' | 'third_party' etc.
                entity_id=doc.subject_id,
                title=f"Document expiring: {doc.document_type}",
                message=(
                    f"Document '{doc.document_type}' "
                    f"(subject: {doc.subject_type} {doc.subject_id}) "
                    f"expires in {days_remaining} day(s) on {doc.expiry_date.isoformat()}."
                ),
                channel="dashboard",
            )
            try:
                await create_alert(db, doc.tenant_id, payload)
                alert_count += 1
            except Exception as exc:
                # Log and continue — do not abort the whole batch for one document
                logger.warning(
                    "task_check_document_expiry_alert_error",
                    doc_id=str(doc.id),
                    error=str(exc),
                )

    logger.info("task_check_document_expiry", alerts_generated=alert_count)
    return f"Generated {alert_count} document expiry alerts"
```

### Task 2 — Unit test for document expiry alert task

```python
async def test_doc_expiry_alerts(async_client, db, tenant, user_token):
    """TP-10: Document expiring in 15 days generates 1 alert with correct fields."""
    from datetime import date, timedelta
    from unittest.mock import AsyncMock, MagicMock, patch
    import uuid

    doc_id = uuid.uuid4()
    subject_id = uuid.uuid4()
    expiry = date.today() + timedelta(days=15)

    # Mock OperationalDocument row returned from admin session
    mock_doc = MagicMock()
    mock_doc.id = doc_id
    mock_doc.tenant_id = tenant["id"]
    mock_doc.subject_type = "driver"
    mock_doc.subject_id = subject_id
    mock_doc.document_type = "license"
    mock_doc.expiry_date = expiry

    captured_alerts = []

    async def mock_create_alert(db, tenant_id, payload):
        captured_alerts.append(payload)

    with patch(
        "app.modules.third_party.models.OperationalDocument",
    ):
        with patch("app.modules.alerts.service.create_alert", side_effect=mock_create_alert):
            # Build minimal ctx as the ARQ worker would have
            from sqlalchemy.ext.asyncio import AsyncSession
            from unittest.mock import AsyncMock

            mock_session = AsyncMock(spec=AsyncSession)
            mock_scalars = MagicMock()
            mock_scalars.scalars.return_value.all.return_value = [mock_doc]
            mock_session.execute = AsyncMock(return_value=mock_scalars)
            mock_session.__aenter__ = AsyncMock(return_value=mock_session)
            mock_session.__aexit__ = AsyncMock(return_value=False)

            mock_factory = MagicMock()
            mock_factory.return_value = mock_session

            ctx = {"db_factory": mock_factory}
            from app.worker import task_check_document_expiry
            result = await task_check_document_expiry(ctx)

    assert len(captured_alerts) == 1
    alert = captured_alerts[0]
    assert alert.entity_type == "driver"
    assert alert.entity_id == subject_id
    assert alert.priority == "high"   # 15 days — not critical (>7)
    assert f"doc_expiry:{doc_id}" in alert.request_reference
    assert expiry.isoformat() in alert.request_reference
```

Note: The test above is a structural integration test. The executor should adapt mocking depth to match what actually runs in the test suite — the key assertions are: 1 alert generated, correct entity_type/entity_id, correct priority for 15-day expiry, request_reference includes doc_id and expiry_date.
