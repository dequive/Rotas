        json=payload,
        headers=extra_headers,

    assert res1.status_code == 201, res1.text
    id1 = res1.json()["id"]

    # Second call with same key — must return same result, no 409
    res2 = await async_client.post(
        f"/api/v1/trips/{trip.id}/advance",
        json=payload,
        headers=extra_headers,
    )
    assert res2.status_code in (200, 201), res2.text
    assert res2.json()["id"] == id1  # same record, no duplicate
