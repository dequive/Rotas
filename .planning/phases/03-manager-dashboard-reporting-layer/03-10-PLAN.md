): Promise<{ jobId: string; status: string }> {
  // export_format is a FastAPI Query param — pass via URL, not request body

**Export polling state (per billing document, not per trip):**
```typescript
const [exportState, setExportState] = useState<Record<string, {
  status: "idle" | "polling" | "done" | "failed";
  jobId?: string;
  format?: "pdf" | "xlsx";
}>>({});

async function handleExport(documentId: string, format: "pdf" | "xlsx") {
  setExportState(prev => ({ ...prev, [documentId + format]: { status: "polling", format } }));
  try {
    const { jobId } = await enqueueExportJob(documentId, format);
    let attempts = 0;
    const poll = setInterval(async () => {
      attempts++;
      if (attempts > 60) {
        clearInterval(poll);
        setExportState(prev => ({ ...prev, [documentId + format]: { status: "failed", jobId, format } }));
        return;
      }
      const result = await getJobStatus(jobId);
      if (result.status === "done") {
        clearInterval(poll);
        setExportState(prev => ({ ...prev, [documentId + format]: { status: "done", jobId, format } }));
      } else if (result.status === "failed") {
        clearInterval(poll);
        setExportState(prev => ({ ...prev, [documentId + format]: { status: "failed", jobId, format } }));
      }
    }, 2000);
  } catch {
    setExportState(prev => ({ ...prev, [documentId + format]: { status: "failed", format } }));
  }
}
```
