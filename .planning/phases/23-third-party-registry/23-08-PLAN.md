## Goal

Implement `PartyDirectoryService.search` as a single `UNION ALL` SQL query across `drivers`, `clients`, and `third_parties` — not three API calls — and expose it as `GET /api/v1/third-party/party-directory` with optional subject_type filtering and pagination.

## Tasks

### Task 1 — UNION ALL service function

**File:** `backend/app/modules/third_party/service.py` — append `search_party_directory` function

Follow the UNION ALL pattern from `RESEARCH.md section 8`. Client `is_active` (boolean) must be cast to a string to align with Driver.status and ThirdParty.status (which are VARCHAR). The `name` column must be normalized across types:
- Driver: use `Driver.full_name`
- Client: use `Client.trading_name`
- ThirdParty: use `ThirdParty.name`

```python
async def search_party_directory(
    db: AsyncSession,
    tenant_id: UUID,
    *,
    query: str | None = None,
    subject_types: list[str] | None = None,
    limit: int = 50,
    offset: int = 0,
) -> list[dict]:
    """Single UNION ALL across drivers/clients/third_parties.

    subject_types: optional list of subject types to filter to.
                   e.g. ['driver'] returns only drivers.
                   None or empty list returns all three types.
    query: optional name search (ILIKE %query%).
    """
    from sqlalchemy import func, literal, String, cast, text, union_all
    from app.modules.drivers.models import Driver
    from app.modules.clients.models import Client

    _all_types = subject_types or ["driver", "client", "third_party"]

    subqueries = []

    if "driver" in _all_types:
        q = select(
            Driver.id.label("subject_id"),
            literal("driver").label("subject_type"),
            Driver.full_name.label("name"),
            Driver.status.label("status"),
        ).where(Driver.tenant_id == tenant_id)
        if query:
            q = q.where(Driver.full_name.ilike(f"%{query}%"))
        subqueries.append(q)

    if "client" in _all_types:
        q = select(
            Client.id.label("subject_id"),
            literal("client").label("subject_type"),
            Client.trading_name.label("name"),
            cast(Client.is_active, String).label("status"),
        ).where(Client.tenant_id == tenant_id)
        if query:
            q = q.where(Client.trading_name.ilike(f"%{query}%"))
        subqueries.append(q)

    if "third_party" in _all_types:
        q = select(
            ThirdParty.id.label("subject_id"),
            literal("third_party").label("subject_type"),
            ThirdParty.name.label("name"),
            ThirdParty.status.label("status"),
        ).where(ThirdParty.tenant_id == tenant_id)
        if query:
            q = q.where(ThirdParty.name.ilike(f"%{query}%"))
        subqueries.append(q)

    if not subqueries:
        return []

    if len(subqueries) == 1:
        stmt = subqueries[0]
    else:
        stmt = union_all(*subqueries)

    # Wrap in a subquery to apply ORDER BY + LIMIT + OFFSET on the UNION result
    paginated = (
        select(text("subject_id, subject_type, name, status"))
        .select_from(stmt.subquery("party_union"))
        .order_by(text("name ASC"))
        .limit(limit)
        .offset(offset)
    )

    result = await db.execute(paginated)
    rows = result.fetchall()
    return [
        {
            "subject_id": row.subject_id,
            "subject_type": row.subject_type,
            "name": row.name,
            "status": row.status,
        }
        for row in rows
    ]
```

@router.get("/party-directory")
async def search_party_directory(
    q: str | None = Query(None, description="Name search string"),
    subject_type: str | None = Query(
        None,
        description="Filter to one subject type: driver | client | third_party"
    ),
    limit: int = Query(50, ge=1, le=200),
    offset: int = Query(0, ge=0),
    db: AsyncSession = Depends(get_session),
    principal=Depends(get_current_principal),
):

    return await service.search_party_directory(
        db,
        principal.tenant_id,
        query=q,
        subject_types=subject_types,
        limit=limit,
        offset=offset,
    )
```

async def test_party_directory(async_client, db, tenant, user_token):
    """TP-11: UNION ALL query returns drivers + clients + third_parties.

    Setup:
    - Create 1 driver via POST /api/v1/drivers
    - Create 1 client via POST /api/v1/clients
    - Create 1 third_party via POST /api/v1/third-party

    Assertions:
    1. GET /api/v1/third-party/party-directory returns 3 items
    2. Each item has: subject_id, subject_type, name, status
    3. GET /api/v1/third-party/party-directory?subject_type=third_party returns 1 item
       with subject_type='third_party'
    4. GET /api/v1/third-party/party-directory?q=<driver_name_fragment> returns 1 item
    """
    headers = {"Authorization": f"Bearer {user_token}"}

    # Create driver (check existing driver create endpoint)
    driver_resp = await async_client.post(
        "/api/v1/drivers",
        json={"full_name": "Américo Machava", "phone": "+258841234567",
              "license_number": "L-TEST-001"},
        headers=headers,
    )
    assert driver_resp.status_code == 201

    # Create client
    client_resp = await async_client.post(
        "/api/v1/clients",
        json={"trading_name": "Transportes Beira Lda", "nuit": "400123456"},
        headers=headers,
    )
    assert client_resp.status_code == 201

    # Create third party
    tp_resp = await async_client.post(
        "/api/v1/third-party",
        json={"name": "Petromoc Maputo", "status": "active"},
        headers=headers,
    )
    assert tp_resp.status_code == 201

    # Query all — expect 3 results
    all_resp = await async_client.get(
        "/api/v1/third-party/party-directory", headers=headers
    )
    assert all_resp.status_code == 200
    data = all_resp.json()
    assert len(data) == 3
    subject_types_found = {row["subject_type"] for row in data}
    assert subject_types_found == {"driver", "client", "third_party"}
    for row in data:
        assert "subject_id" in row
        assert "name" in row
        assert "status" in row

    # Filter by subject_type=third_party — expect 1 result
    tp_only = await async_client.get(
        "/api/v1/third-party/party-directory?subject_type=third_party", headers=headers
    )
    assert tp_only.status_code == 200
    assert len(tp_only.json()) == 1
    assert tp_only.json()[0]["subject_type"] == "third_party"

    # Name search
    search = await async_client.get(
        "/api/v1/third-party/party-directory?q=Américo", headers=headers
    )
    assert search.status_code == 200
    assert len(search.json()) == 1
    assert search.json()[0]["subject_type"] == "driver"
```
