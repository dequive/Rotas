Create apps/manager/app/components/ClientCombobox.tsx as a client component ("use client").

```typescript
"use client";

import { useState, useEffect } from "react";
import { Check, ChevronsUpDown } from "lucide-react";
import { cn } from "@/lib/utils";
import { Button } from "@/components/ui/button";
import {
  Command, CommandEmpty, CommandGroup, CommandInput, CommandItem, CommandList,
} from "@/components/ui/command";
import { Popover, PopoverContent, PopoverTrigger } from "@/components/ui/popover";
import { Skeleton } from "@/components/ui/skeleton";

interface Client {
  id: string;
  trading_name: string;
  nuit: string;
}

interface ClientComboboxProps {
  value?: string;              // selected client_id UUID
  onChange: (clientId: string, tradingName: string) => void;
  disabled?: boolean;
}

export function ClientCombobox({ value, onChange, disabled }: ClientComboboxProps) {
  const [open, setOpen] = useState(false);
  const [clients, setClients] = useState<Client[]>([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const selectedClient = clients.find((c) => c.id === value);

  useEffect(() => {
    if (!open) return;
    setLoading(true);
    setError(null);
    // Read auth headers from cookie/localStorage — same pattern as other client components
    const token = localStorage.getItem("rotas_access_token") || "";
    const tenantId = localStorage.getItem("rotas_tenant_id") || "";
    fetch("/api/v1/clients?limit=200", {
      headers: {
        Authorization: `Bearer ${token}`,
        "X-Tenant-Id": tenantId,
      },
    })
      .then((r) => {
        if (!r.ok) throw new Error("Erro ao carregar clientes");
        return r.json();
      })
      .then((data) => {
        setClients(Array.isArray(data) ? data : []);
      })
      .catch(() => setError("Erro ao carregar clientes"))
      .finally(() => setLoading(false));
  }, [open]);

  return (
    <Popover open={open} onOpenChange={setOpen}>
      <PopoverTrigger asChild>
        <Button
          variant="outline"
          role="combobox"
          aria-expanded={open}
          aria-haspopup="listbox"
          className="w-full justify-between min-h-[38px]"
          disabled={disabled}
        >
          {selectedClient ? selectedClient.trading_name : "Pesquisar cliente…"}
          <ChevronsUpDown className="ml-2 h-4 w-4 shrink-0 opacity-50" />
        </Button>
      </PopoverTrigger>
      {selectedClient && (
        <p className="text-[11px] font-mono text-[var(--muted-color)] mt-1">
          NUIT: {selectedClient.nuit}
        </p>
      )}
      <PopoverContent className="w-[320px] p-0">
        <Command>
          <CommandInput placeholder="Pesquisar cliente…" />
          <CommandList>
            {error && (
              <p className="text-[var(--error)] text-xs p-3">{error}</p>
            )}
            {loading && !error && (
              <div className="p-2 space-y-2">
                {[1, 2, 3].map((i) => (
                  <Skeleton key={i} className="h-8 w-full" />
                ))}
              </div>
            )}
            {!loading && !error && (
              <>
                <CommandEmpty>
                  Nenhum cliente encontrado. Crie um cliente primeiro.
                </CommandEmpty>
                <CommandGroup>
                  {clients.map((client) => (
                    <CommandItem
                      key={client.id}
                      value={client.trading_name}
                      onSelect={() => {
                        onChange(client.id, client.trading_name);
                        setOpen(false);
                      }}
                    >
                      <Check
                        className={cn(
                          "mr-2 h-4 w-4",
                          value === client.id ? "opacity-100" : "opacity-0"
                        )}
                      />
                      <span>{client.trading_name}</span>
                      <span className="ml-auto text-[11px] font-mono text-[var(--muted-color)]">
                        {client.nuit}
                      </span>
                    </CommandItem>
                  ))}
                </CommandGroup>
              </>
            )}
          </CommandList>
        </Command>
      </PopoverContent>
    </Popover>
  );
}
```

Note on auth: The fetch uses `localStorage.getItem("rotas_access_token")` which is the established driver PWA pattern. Verify the manager app uses the same localStorage key by reading `apps/manager/app/lib/auth.ts` before implementing — if the manager uses HttpOnly cookies instead, use `credentials: "include"` instead of the Authorization header.
