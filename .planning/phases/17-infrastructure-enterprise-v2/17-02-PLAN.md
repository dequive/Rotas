```python
async def task_update_active_tenants_metric(ctx: dict) -> str:
    """INFRA2-03: Update Prometheus active_tenants gauge every 5 minutes.

    Uses admin session (BYPASSRLS) to count across all tenants.
    """
    from sqlalchemy import select, func
    from app.modules.tenants.models import Tenant

    async with get_admin_session() as db:
        result = await db.execute(
            select(func.count()).where(Tenant.is_active == True)
        )
        count = result.scalar_one_or_none() or 0

    # Update gauge if prometheus_client is available
    try:
        from prometheus_client import Gauge
        # Use a named gauge (must match name registered in main.py)
        g = Gauge(
            "rotas_active_tenants_total",
            "Number of active tenants in the platform",
        )
        g.set(count)
    except Exception:
        pass  # prometheus_client may not be available in all envs

    logger.info("task_update_active_tenants_metric", active_tenants=count)
    return f"Active tenants: {count}"
```

```python
async def task_worker_heartbeat(ctx: dict) -> str:
    """INFRA2-04: Worker heartbeat — update Redis key every 30 seconds.

    Key: arq:health:worker_heartbeat
    Value: ISO timestamp
    TTL: 90 seconds (3x the heartbeat interval — dead after 3 missed beats)

    The /health/deep endpoint reads this key to determine if the ARQ worker is alive.
    """
    from datetime import datetime, timezone

    now = datetime.now(timezone.utc).isoformat()

    # Use the arq Redis connection from ctx
    redis = ctx.get("redis")
    if redis:
        await redis.setex("arq:health:worker_heartbeat", 90, now)
        logger.info("task_worker_heartbeat", timestamp=now)
        return f"Heartbeat: {now}"
    else:
        logger.warning("task_worker_heartbeat_no_redis")
        return "No Redis — heartbeat skipped"
```

Registar em `WorkerSettings.cron_jobs`:
```python
# INFRA2-04: Every 30 seconds
cron(task_worker_heartbeat, second=30),
# Note: ARQ cron minimum is 1 minute in some versions — if so, use:
# cron(task_worker_heartbeat, minute={0,1,2,...59})  — every minute
# Check arq version: if < 0.25, use minute-based
```

Registar em `functions`:
```python

```

```python
@app.get("/health/deep", tags=["operation"])
async def health_deep(request: Request) -> dict:
    """
    INFRA2-04: Deep health check — verifies DB, Redis, and ARQ worker.
    Returns HTTP 200 if all components healthy, HTTP 503 if any critical component fails.

    Use this endpoint for alerting and monitoring (Grafana, Datadog, etc.)
    NOT for Railway health check (use /health for that — it's always 200).
    """
    import asyncio
    from datetime import datetime, timezone, timedelta
    from fastapi.responses import JSONResponse
    from sqlalchemy import text

    checks: dict[str, str] = {}
    overall_ok = True

    # 1. Database check
    try:
        async with request.app.state.engine.begin() as conn:
            await asyncio.wait_for(
                conn.execute(text("SELECT 1")),
                timeout=2.0,
            )
        checks["db"] = "ok"
    except Exception as e:
        checks["db"] = f"error: {type(e).__name__}"
        overall_ok = False

    # 2. Redis check
    redis = getattr(request.app.state, "redis", None)
    if redis:
        try:
            await asyncio.wait_for(redis.ping(), timeout=1.0)
            checks["redis"] = "ok"
        except Exception as e:
            checks["redis"] = f"error: {type(e).__name__}"
            overall_ok = False
    else:
        checks["redis"] = "not_configured"
        # Redis is required in production — mark unhealthy
        if settings.environment == "production":
            overall_ok = False

    # 3. ARQ worker check (reads heartbeat key set by task_worker_heartbeat)
    try:
        if redis:
            heartbeat = await redis.get("arq:health:worker_heartbeat")
            if heartbeat:
                last_beat = datetime.fromisoformat(heartbeat)
                age_seconds = (datetime.now(timezone.utc) - last_beat).total_seconds()
                if age_seconds < 90:
                    checks["arq_worker"] = f"ok (last beat {int(age_seconds)}s ago)"
                else:
                    checks["arq_worker"] = f"stale (last beat {int(age_seconds)}s ago)"
                    overall_ok = False
            else:
                checks["arq_worker"] = "no_heartbeat (worker may be starting)"
                # Don't mark unhealthy on startup — give worker 90s to send first heartbeat
        else:
            checks["arq_worker"] = "unknown (no Redis)"
    except Exception as e:
        checks["arq_worker"] = f"error: {type(e).__name__}"
        overall_ok = False

    result = {
        "status": "ok" if overall_ok else "degraded",
        "checks": checks,
        "timestamp": datetime.now(timezone.utc).isoformat(),
    }

    if overall_ok:
        return result
    else:
        return JSONResponse(status_code=503, content=result)
```
