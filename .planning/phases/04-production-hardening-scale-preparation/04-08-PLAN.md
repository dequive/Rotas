**Step 1: Add RLS context variable and event listener**

Add the following to `backend/app/database.py` after the engine and AsyncSessionLocal definitions:

```python
from contextvars import ContextVar

from sqlalchemy import event, text

# RLS context variable — holds current request's tenant_id
# ContextVar is asyncio-safe (no shared state between coroutines)
_rls_tenant: ContextVar[str | None] = ContextVar("_rls_tenant", default=None)


def set_rls_tenant(tenant_id: str | None) -> None:
    """Set the current request's tenant_id for RLS enforcement.

    Called by get_session() dependency before yielding the session.
    Call with None to clear after request completes.
    D-17: Uses SET LOCAL (transaction-scoped, not connection-scoped) — critical for pool safety.
    """
    _rls_tenant.set(tenant_id)


@event.listens_for(AsyncSessionLocal.sync_session_class, "after_begin")
def _inject_rls_tenant(session, transaction, connection):
    """SQLAlchemy after_begin event: fires once per transaction.

    Executes SET LOCAL app.tenant_id so PostgreSQL RLS policy can read it.
    SET LOCAL (not SET) is mandatory — scoped to current transaction only.
    With asyncpg connection pooling, the connection is reused across requests.
    SET LOCAL ensures tenant_id does not leak across request boundaries.
    """
    tid = _rls_tenant.get()
    if tid is not None:
        connection.execute(
            text(f"SET LOCAL app.tenant_id = '{tid}'")  # noqa: S608 — controlled input from JWT
        )
```

**Step 2: Update get_session() to inject tenant_id from the request**

The challenge: `get_session()` must know the tenant_id before the session opens, but it is called via `Depends()` — it can read from `_rls_tenant` which should be set by the auth dependency.

The canonical approach: auth dependencies run before get_session sets the tenant. We use a different pattern — `get_session` accepts the principal as a parameter:

```python
async def get_session(
    principal=Depends(get_current_principal),
) -> AsyncIterator[AsyncSession]:
    """FastAPI dependency: yields AsyncSession with RLS tenant_id pre-set.

    Sets the request-scoped ContextVar before opening the session.
    Clears it on exit so pool connections are not contaminated.
    """
    set_rls_tenant(str(principal.tenant_id))
    try:
        async with AsyncSessionLocal() as session:
            yield session
    finally:
        set_rls_tenant(None)
```

**IMPORTANT CAVEAT:** The existing `get_session()` takes no parameters. Changing its signature to `Depends(get_current_principal)` means:
1. All unauthenticated routes that use `get_session` (if any) will break
2. The `get_current_principal` dependency itself uses `get_session` — this would create a circular dependency

**Correct approach to avoid circular dependency:**

Keep a separate `get_session_raw()` for use by auth dependencies and unauthenticated routes:

```python
# Existing signature preserved for auth/unauthenticated use
async def get_session_raw() -> AsyncIterator[AsyncSession]:
    """Raw session without RLS tenant injection. Used by auth module only."""
    async with AsyncSessionLocal() as session:
        yield session


# RLS-enabled session for all tenant-aware routes
async def get_session(
    principal=Depends(get_current_principal),
) -> AsyncIterator[AsyncSession]:
    """Session with RLS tenant_id injected. Use for all tenant-aware endpoints."""
    set_rls_tenant(str(principal.tenant_id))
    try:
        async with AsyncSessionLocal() as session:
            yield session
    finally:
        set_rls_tenant(None)
```

```python
"""add_rls_policies

D-16 through D-19: PostgreSQL Row Level Security for multitenant isolation.
Creates rotas_app (RLS enforced) and rotas_admin (BYPASSRLS) roles.
Enables RLS on all tenant-scoped tables.
Creates tenant_isolation policy using SET LOCAL app.tenant_id.
"""
from alembic import op

TENANT_SCOPED_TABLES = [
    "users", "drivers", "driver_devices", "driver_sessions",
    "vehicles", "fuel_logs", "fuel_tanks", "fuel_purchases", "fuel_receipts",
    "fuel_movements", "vehicle_refuels", "fuel_stock_counts",
    "trips", "trip_stops", "trip_costs", "dispatch_clearances",
    "trip_execution_events", "trip_incidents", "known_routes", "trip_orders",
    "checklists", "checklist_templates", "checklist_responses",
    "load_permits", "cargo_manifests", "transport_documents", "delivery_proofs",
    "billing_documents", "billing_items", "contracts",
    "maintenance_requests", "work_orders", "work_order_tasks",
    "spare_parts_inventory", "spare_part_movements", "maintenance_parts_used",
    "workshop_tools", "tool_checkouts", "maintenance_plans", "maintenance_schedule",
    "operational_exceptions", "alerts", "sync_events", "idempotency_keys", "audit_logs",
]


def upgrade() -> None:
    # D-18: Create application roles
    # Use IF NOT EXISTS to make migration idempotent
    op.execute("DO $$ BEGIN CREATE ROLE rotas_app; EXCEPTION WHEN duplicate_object THEN NULL; END $$")
    op.execute("DO $$ BEGIN CREATE ROLE rotas_admin BYPASSRLS; EXCEPTION WHEN duplicate_object THEN NULL; END $$")

    # Grant rotas_app permissions on all tables
    op.execute(
        "GRANT SELECT, INSERT, UPDATE, DELETE "
        "ON ALL TABLES IN SCHEMA public TO rotas_app"
    )
    op.execute(
        "GRANT USAGE ON ALL SEQUENCES IN SCHEMA public TO rotas_app"
    )

    # D-19: Enable RLS and create tenant isolation policy on all tenant-scoped tables
    for table in TENANT_SCOPED_TABLES:
        op.execute(f"ALTER TABLE {table} ENABLE ROW LEVEL SECURITY")
        op.execute(f"ALTER TABLE {table} FORCE ROW LEVEL SECURITY")
        # D-17: current_setting 'true' arg returns NULL if not set (no error)
        # Cast tenant_id::text to match current_setting text type (Pitfall 6)
        op.execute(f"""
            CREATE POLICY tenant_isolation ON {table}
            USING (tenant_id::text = current_setting('app.tenant_id', true))
        """)


def downgrade() -> None:
    for table in TENANT_SCOPED_TABLES:
        op.execute(f"DROP POLICY IF EXISTS tenant_isolation ON {table}")
        op.execute(f"ALTER TABLE {table} DISABLE ROW LEVEL SECURITY")
    # Note: roles are NOT dropped in downgrade to avoid breaking existing connections
```

```python
from sqlalchemy import text

async def test_rls_tenant_isolation_policy_exists():
    """D-19: pg_policies shows tenant_isolation policy on trips table."""
    async with AsyncSessionLocal() as db:
        result = await db.execute(
            text(
                "SELECT 1 FROM pg_policies "
                "WHERE tablename = 'trips' AND policyname = 'tenant_isolation'"
            )
        )
    assert result.scalar() is not None, "RLS policy 'tenant_isolation' not found on trips table"


async def test_rls_set_local_scoped_to_transaction():
    """D-17: SET LOCAL app.tenant_id is transaction-scoped, not session-scoped."""
    from uuid import uuid4
    tid = str(uuid4())
    async with AsyncSessionLocal() as db:
        async with db.begin():
            await db.execute(text(f"SET LOCAL app.tenant_id = '{tid}'"))
            result = await db.execute(text("SELECT current_setting('app.tenant_id', true)"))
            in_tx = result.scalar()
        # After transaction ends, setting should be cleared
        result2 = await db.execute(text("SELECT current_setting('app.tenant_id', true)"))
        after_tx = result2.scalar()
    assert in_tx == tid, f"Expected {tid}, got {in_tx}"
    # After transaction, SET LOCAL should be gone (returns empty string or None)
    assert after_tx != tid, "SET LOCAL leaked past transaction boundary — pool safety risk"


async def test_rls_blocks_cross_tenant_trip_access():
    """D-19: With app.tenant_id=A, queries on trips table cannot see tenant B's rows."""
    from uuid import uuid4
    from app.modules.tenants.models import Tenant
    from app.modules.trips.models import Trip
    from app.modules.vehicles.models import Vehicle
    from app.modules.drivers.models import Driver

    # Create two tenants and a trip for each
    suffix = uuid4().hex[:6]
    async with AsyncSessionLocal() as db:
        # Create tenants
        t_a = Tenant(name=f"RLS_A_{suffix}", slug=f"rls-a-{suffix}")
        t_b = Tenant(name=f"RLS_B_{suffix}", slug=f"rls-b-{suffix}")
        db.add_all([t_a, t_b])
        await db.flush()

        # Create minimal vehicle + driver for tenant_b to seed a trip
        v_b = Vehicle(tenant_id=t_b.id, plate=f"RLS-B-{suffix}", category="ligeiro", fuel_type="gasolina")
        d_b = Driver(tenant_id=t_b.id, full_name=f"Driver B {suffix}", phone=f"25881{suffix}")
        db.add_all([v_b, d_b])
        await db.flush()

        trip_b = Trip(
            tenant_id=t_b.id,
            vehicle_id=v_b.id,
            driver_id=d_b.id,
            origin="Maputo",
            destination="Beira",
            status="planned",
        )
        db.add(trip_b)
        await db.commit()
        t_a_id = t_a.id
        t_b_id = t_b.id
        trip_b_id = trip_b.id

    # Now query as tenant_A — should NOT see tenant_B's trip
    set_rls_tenant(str(t_a_id))
    try:
        async with AsyncSessionLocal() as db:
            from sqlalchemy import select
            result = await db.execute(
                select(Trip).where(Trip.id == trip_b_id)
            )
            row = result.scalar()
    finally:
        set_rls_tenant(None)

    assert row is None, (
        f"RLS FAILED: tenant A can see tenant B's trip {trip_b_id}. "
        "Cross-tenant data leak detected."
    )
```
