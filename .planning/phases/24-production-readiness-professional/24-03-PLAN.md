```typescript
import { apiFetch } from "./api";

export interface ThirdParty {
  id: string;
  name: string;
  tax_id: string | null;
  status: string;
  roles: string[];
  sector: string | null;
  activity_code: string | null;
  average_score: string | null;
}

export interface ThirdPartyContact {
  id: string;
  name: string;
  role: string | null;
  phone: string | null;
  email: string | null;
  is_primary: boolean;
  created_at: string;
}

export interface LedgerEntry {
  id: string;
  entry_type: "debit" | "credit";
  amount: string;
  source_type: string;
  description: string | null;
  entry_date: string;
}

export interface SupplierAccount {
  third_party_id: string;
  total_debits: string;
  total_credits: string;
  balance: string;
  entries: LedgerEntry[];
}

export interface EvaluationCriterion {
  name: string;
  weight: number;
  score: number;
}

export interface SupplierEvaluation {
  id: string;
  evaluation_date: string;
  criteria: EvaluationCriterion[];
  score: string;
  notes: string | null;
  evaluated_by: string | null;
}

export interface EvaluationsResult {
  average_score: string | null;
  evaluations: SupplierEvaluation[];
}

export async function loadThirdParties(params?: {
  role_type?: string;
  status?: string;
  sector?: string;
  limit?: number;
  offset?: number;
}): Promise<{ items: ThirdParty[]; total: number }> {
  const qs = new URLSearchParams();
  if (params?.role_type) qs.set("role_type", params.role_type);
  if (params?.status) qs.set("status", params.status);
  if (params?.sector) qs.set("sector", params.sector);
  if (params?.limit) qs.set("limit", String(params.limit));
  if (params?.offset) qs.set("offset", String(params.offset));
  const query = qs.toString() ? `?${qs}` : "";
  return apiFetch(`/api/v1/third-party${query}`);
}

export async function loadThirdParty(id: string): Promise<ThirdParty> {
  return apiFetch(`/api/v1/third-party/${id}`);
}

export async function loadContacts(thirdPartyId: string): Promise<ThirdPartyContact[]> {
  return apiFetch(`/api/v1/third-party/${thirdPartyId}/contacts`);
}

export async function loadSupplierAccount(thirdPartyId: string): Promise<SupplierAccount> {
  return apiFetch(`/api/v1/third-party/${thirdPartyId}/account`);
}

export async function loadEvaluations(thirdPartyId: string): Promise<EvaluationsResult> {
  return apiFetch(`/api/v1/third-party/${thirdPartyId}/evaluations`);
}
```

**File:** `apps/manager/app/api/third-party/route.ts`

**Action:** CREATE

**Details:**
Next.js Route Handler that proxies GET requests to the backend (needed so Client Components can fetch third-party lists without exposing httpOnly cookies):

```typescript
import { NextRequest, NextResponse } from "next/server";
import { requireSession } from "@/app/lib/auth";

const API_BASE = process.env.NEXT_PUBLIC_API_URL ?? "http://localhost:8000";

export async function GET(request: NextRequest) {
  const session = await requireSession();
  const { searchParams } = new URL(request.url);
  const query = searchParams.toString() ? `?${searchParams}` : "";
  const res = await fetch(`${API_BASE}/api/v1/third-party${query}`, {
    headers: {
      Authorization: `Bearer ${session.accessToken}`,
      "X-Tenant-Id": session.tenantId,
    },
  });
  const data = await res.json();
  return NextResponse.json(data, { status: res.status });
}
```

```tsx
import { requireSession } from "@/app/lib/auth";
import { loadThirdParties } from "@/app/lib/third-party-api";
import { SidebarLayout } from "@/app/components/SidebarLayout";
import { StatusBadge } from "@/app/components/ui/StatusBadge";
import Link from "next/link";
import { Building2 } from "lucide-react";

export default async function TerceirosPage({
  searchParams,
}: {
  searchParams: { role_type?: string; status?: string; sector?: string; page?: string };
}) {
  await requireSession();
  const offset = ((Number(searchParams.page ?? 1) - 1) * 50);
  const result = await loadThirdParties({
    role_type: searchParams.role_type,
    status: searchParams.status,
    sector: searchParams.sector,
    limit: 50,
    offset,
  });
  const items = Array.isArray(result) ? result : (result as { items?: unknown[] }).items ?? result;

  return (
    <SidebarLayout active="terceiros">
      <div className="page-header">
        <div>
          <h1>Terceiros</h1>

        </div>
        {/* "Novo Terceiro" button — links to creation form or opens modal */}
        <a href="/terceiros/novo" className="btn-primary flex items-center gap-2">
          <Building2 size={16} /> Novo Terceiro
        </a>
      </div>

      {/* Filter bar */}
      <form method="GET" className="panel flex flex-wrap gap-3 items-end mb-4">
        <div>
          <label className="label-xs">Tipo</label>
          <select name="role_type" defaultValue={searchParams.role_type ?? ""}
                  className="input-sm">
            <option value="">Todos</option>
            <option value="supplier">Fornecedor</option>
            <option value="service_provider">Prestador</option>
            <option value="client">Cliente</option>
          </select>
        </div>
        <div>
          <label className="label-xs">Estado</label>
          <select name="status" defaultValue={searchParams.status ?? ""}
                  className="input-sm">
            <option value="">Todos</option>
            <option value="active">Activo</option>
            <option value="inactive">Inactivo</option>
            <option value="suspended">Suspenso</option>
          </select>
        </div>
        <div>
          <label className="label-xs">Sector</label>
          <input name="sector" type="text" defaultValue={searchParams.sector ?? ""}
                 placeholder="Filtrar sector..." className="input-sm" />
        </div>
        <button type="submit" className="btn-secondary">Filtrar</button>
      </form>

      <section className="panel">
        <div className="table-wrap">
          <table className="table">
            <thead>
              <tr>
                <th>Nome</th>
                <th>Tipo</th>
                <th>Sector</th>
                <th>Estado</th>
                <th>Score médio</th>
                <th>Acções</th>
              </tr>
            </thead>
            <tbody>
              {(Array.isArray(items) ? items : []).map((tp: Record<string, unknown>) => (
                <tr key={String(tp.id)} className="hover:bg-surface-2 cursor-pointer">
                  <td className="font-medium">{String(tp.name ?? "")}</td>
                  <td>
                    {(tp.roles as string[] | undefined)?.map((r: string) => (
                      <span key={r} className="badge badge-neutral mr-1">
                        <span className="dot" />
                        {r === "supplier" ? "Fornecedor"
                         : r === "service_provider" ? "Prestador"
                         : r === "client" ? "Cliente" : r}
                      </span>
                    ))}
                  </td>
                  <td className="text-muted">{String(tp.sector ?? "—")}</td>
                  <td>
                    <StatusBadge
                      tone={tp.status === "active" ? "green"
                             : tp.status === "suspended" ? "orange" : "red"}
                      label={tp.status === "active" ? "Activo"
                              : tp.status === "suspended" ? "Suspenso" : "Inactivo"}
                    />
                  </td>
                  <td className="font-mono text-sm">
                    {tp.average_score ? String(tp.average_score) : "—"}
                  </td>
                  <td>
                    <Link href={`/terceiros/${String(tp.id)}`}
                          className="btn-ghost btn-sm">Ver</Link>
                  </td>
                </tr>
              ))}
              {(Array.isArray(items) ? items : []).length === 0 && (
                <tr>
                  <td colSpan={6} className="text-center text-muted py-8">
                    Nenhum terceiro encontrado
                  </td>
                </tr>
              )}
            </tbody>
          </table>
        </div>

    </SidebarLayout>
```

### Task 3: Terceiros detail page with 5 tabs

**File:** `apps/manager/app/terceiros/[id]/page.tsx`

**Action:** CREATE

**Details:**
Server Component that loads all data concurrently then renders a tabbed layout. Tabs are rendered as a `<Tabs>` Client Component (same Radix pattern used in `apps/manager/app/manutencao/page.tsx`).

```tsx
import { requireSession } from "@/app/lib/auth";
import {
  loadThirdParty, loadContacts, loadSupplierAccount, loadEvaluations
} from "@/app/lib/third-party-api";
import { loadOperationalDocuments } from "@/app/lib/third-party-api";
// loadOperationalDocuments: apiFetch(`/api/v1/third-party/documents?subject_type=third_party&subject_id=${id}`)
// Add this function to third-party-api.ts
import { SidebarLayout } from "@/app/components/SidebarLayout";
import { StatusBadge } from "@/app/components/ui/StatusBadge";
import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";
import { notFound } from "next/navigation";

export default async function TerceiroDetailPage({ params }: { params: { id: string } }) {
  await requireSession();

  // Concurrent data loading — do not await sequentially
  const [party, contacts, account, evalsResult] = await Promise.all([
    loadThirdParty(params.id).catch(() => null),
    loadContacts(params.id).catch(() => []),
    loadSupplierAccount(params.id).catch(() => null),
    loadEvaluations(params.id).catch(() => ({ average_score: null, evaluations: [] })),
  ]);

  if (!party) notFound();

  const balance = account?.balance ?? "0.00";
  const balanceNum = parseFloat(balance);

  return (
    <SidebarLayout active="terceiros">
      {/* Header */}
      <div className="page-header mb-6">
        <div>
          <h1 className="text-2xl font-bold">{party.name}</h1>
          <div className="flex items-center gap-3 mt-1">
            <StatusBadge
              tone={party.status === "active" ? "green"
                     : party.status === "suspended" ? "orange" : "red"}
              label={party.status === "active" ? "Activo"
                      : party.status === "suspended" ? "Suspenso" : "Inactivo"}
            />
            {evalsResult.average_score && (
              <span className="text-sm text-muted font-mono">
                Score: {evalsResult.average_score}/10
              </span>
            )}
          </div>
        </div>
        <a href={`/terceiros/${params.id}/editar`} className="btn-secondary">Editar</a>
      </div>

      {/* Tabs */}
      <Tabs defaultValue="info">
        <TabsList>
          <TabsTrigger value="info">Info</TabsTrigger>
          <TabsTrigger value="contactos">
            Contactos {contacts.length > 0 && `(${contacts.length})`}
          </TabsTrigger>
          <TabsTrigger value="documentos">Documentos</TabsTrigger>
          <TabsTrigger value="conta">Conta Corrente</TabsTrigger>
          <TabsTrigger value="avaliacoes">
            Avaliações {evalsResult.evaluations.length > 0 && `(${evalsResult.evaluations.length})`}
          </TabsTrigger>
        </TabsList>

        {/* Tab: Info */}
        <TabsContent value="info">
          <section className="panel mt-4">
            <h2 className="text-lg font-semibold mb-4">Identidade</h2>
            <dl className="grid grid-cols-2 gap-x-8 gap-y-3 text-sm">
              <dt className="text-muted">NUIT / Tax ID</dt>
              <dd className="font-mono">{party.tax_id ?? "—"}</dd>
              <dt className="text-muted">Sector</dt>
              <dd>{party.sector ?? "—"}</dd>
              <dt className="text-muted">Código de actividade</dt>
              <dd className="font-mono">{party.activity_code ?? "—"}</dd>
              <dt className="text-muted">Roles</dt>
              <dd className="flex flex-wrap gap-1">
                {party.roles?.map((r) => (
                  <span key={r} className="badge badge-neutral">
                    <span className="dot" />
                    {r === "supplier" ? "Fornecedor"
                     : r === "service_provider" ? "Prestador"
                     : r === "client" ? "Cliente" : r}
                  </span>
                ))}
              </dd>
            </dl>
          </section>
        </TabsContent>

        {/* Tab: Contactos */}
        <TabsContent value="contactos">
          <section className="panel mt-4">
            <div className="flex justify-between items-center mb-4">
              <h2 className="text-lg font-semibold">Sub-contactos</h2>
              {/* AddContactModal is a Client Component — stub with link for now */}
              <a href={`/terceiros/${params.id}/contactos/novo`} className="btn-primary btn-sm">
                Adicionar Contacto
              </a>
            </div>
            {contacts.length === 0 ? (
              <p className="text-muted text-sm">Sem contactos registados</p>
            ) : (
              <div className="table-wrap">
                <table className="table">
                  <thead>
                    <tr>
                      <th>Nome</th>
                      <th>Função</th>
                      <th>Telefone</th>
                      <th>Email</th>
                      <th>Principal</th>
                    </tr>
                  </thead>
```
