```python
#!/usr/bin/env python
"""One-shot script: migrate all local File records to Cloudflare R2.

Operator workflow (D-09, D-12):
  1. Configure R2_BUCKET, R2_ENDPOINT_URL, R2_ACCESS_KEY_ID, R2_SECRET_ACCESS_KEY in env
  2. Run: python backend/scripts/migrate_files_to_r2.py
  3. Verify exit code 0 and "0 failed" in summary
  4. Run gate: SELECT count(*) FROM files WHERE storage_provider IN ('local', 'local_stub')
  5. THEN set STORAGE_PROVIDER=R2 in Railway

CRITICAL: Never set STORAGE_PROVIDER=R2 before this script completes successfully (D-09).
"""
import asyncio
import sys
from pathlib import Path
from uuid import UUID

# Allow running from project root: python backend/scripts/migrate_files_to_r2.py
sys.path.insert(0, str(Path(__file__).parent.parent))

from sqlalchemy import select, update

import aiobotocore.session

from app.config import get_settings
from app.database import get_session_raw
from app.modules.files.models import File

_settings = get_settings()

LOCAL_PROVIDERS = ("local", "local_stub")


async def _upload_to_r2(storage_key: str, content: bytes, mime_type: str) -> None:
    session = aiobotocore.session.get_session()
    async with session.create_client(
        "s3",
        endpoint_url=_settings.r2_endpoint_url,
        aws_access_key_id=_settings.r2_access_key_id,
        aws_secret_access_key=_settings.r2_secret_access_key.get_secret_value(),
    ) as client:
        await client.put_object(
            Bucket=_settings.r2_bucket,
            Key=storage_key,
            Body=content,
            ContentType=mime_type,
        )


async def main() -> int:
    uploaded = 0
    failed = []

    async with get_session_raw() as db:
        result = await db.execute(
            select(File).where(File.storage_provider.in_(list(LOCAL_PROVIDERS)))
        )
        files = result.scalars().all()
        total = len(files)
        print(f"Found {total} local file(s) to migrate.")

        for file in files:
            # Reconstruct local disk path
            relative_key = file.storage_key.split("/", maxsplit=1)[1] if "/" in file.storage_key else file.storage_key
            local_path = Path(_settings.local_upload_dir) / relative_key

            try:
                if not local_path.exists():
                    raise FileNotFoundError(f"File not found on disk: {local_path}")

                content = local_path.read_bytes()
                await _upload_to_r2(file.storage_key, content, file.mime_type)

                # Update record
                file.storage_provider = "r2"
                await db.flush()
                uploaded += 1
                print(f"  [OK] {file.id} — {file.storage_key}")

            except Exception as exc:
                failed.append((str(file.id), str(exc)))
                print(f"  [FAIL] {file.id} — {exc}")

        await db.commit()

    print(f"\nMigration complete: {uploaded} uploaded | {len(failed)} failed | {total - uploaded - len(failed)} skipped")

    if failed:
        print("\nFailed files:")
        for file_id, reason in failed:
            print(f"  {file_id}: {reason}")
        return 1
    return 0


if __name__ == "__main__":
    exit_code = asyncio.run(main())
    sys.exit(exit_code)
```
