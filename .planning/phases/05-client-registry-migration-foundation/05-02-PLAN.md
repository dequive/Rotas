Revision ID: c3d4e5f6a7b8
Revises: b2c3d4e5f6a7
Create Date: 2026-06-19

Backfill strategy:
1. Read DISTINCT (tenant_id, lower(trim(client_name))) from contracts
2. INSERT one client per normalized name per tenant
3. UPDATE contracts.client_id WHERE lower(trim(client_name)) matches
4. UPDATE billing_documents.client_id via their contract's client_id

Pre-migration audit query (run before applying this migration):
  SELECT tenant_id, lower(trim(client_name)) AS normalized, count(*),
         array_agg(DISTINCT client_name ORDER BY client_name) AS variants
  FROM contracts
  WHERE client_name IS NOT NULL AND client_name != ''
  GROUP BY 1, 2 HAVING count(*) > 1 ORDER BY 1, 3 DESC;

[Document audit results here before committing this file]
"""

from alembic import op
import sqlalchemy as sa

revision = "c3d4e5f6a7b8"
down_revision = "b2c3d4e5f6a7"
branch_labels = None
depends_on = None


def upgrade() -> None:
    conn = op.get_bind()

    # Step 1: Get distinct normalized (tenant_id, client_name) from contracts
    rows = conn.execute(sa.text("""
        SELECT DISTINCT
            tenant_id,
            lower(trim(client_name)) AS client_name_normalized,
            (
                SELECT client_name FROM contracts c2
                WHERE c2.tenant_id = c1.tenant_id
                  AND lower(trim(c2.client_name)) = lower(trim(c1.client_name))
                  AND c2.client_name IS NOT NULL
                ORDER BY length(c2.client_name) DESC
                LIMIT 1
            ) AS trading_name
        FROM contracts c1
        WHERE client_name IS NOT NULL AND client_name != ''
    """)).fetchall()

    # Step 2: INSERT one client per (tenant_id, normalized name)
    # Use ON CONFLICT DO NOTHING — idempotent re-runs safe
    for row in rows:
        client_id = str(_uuid.uuid4())
        conn.execute(sa.text("""
            INSERT INTO clients (id, tenant_id, trading_name, nuit, payment_terms_days, is_active, created_at, updated_at)
            VALUES (
                :id::uuid,
                :tenant_id,
                :trading_name,
                '000000000',
                30,
                true,
                NOW(),
                NOW()
            )
            ON CONFLICT (tenant_id, nuit) DO NOTHING
        """), {
            "id": client_id,
            "tenant_id": str(row.tenant_id),
            "trading_name": row.trading_name or row.client_name_normalized,
        })

    # NOTE: The above uses a placeholder NUIT '000000000' because contracts did not store NUIT
    # per-client historically. Phase 5 UI requires managers to complete NUITs manually.
    # The unique constraint is on (tenant_id, nuit) — a single placeholder is used per tenant
    # only if no real NUIT is available. Managers can PATCH clients to set real NUITs.
    # If contracts have client_nuit populated (Phase 15), use that instead — see below.

    # Better: use client_nuit from contracts when available
    # (This replaces the above if client_nuit is populated in your dataset)
    conn.execute(sa.text("""
        INSERT INTO clients (id, tenant_id, trading_name, nuit, payment_terms_days, is_active, created_at, updated_at)
        SELECT DISTINCT ON (tenant_id, lower(trim(client_name)))
            gen_random_uuid(),
            tenant_id,
            -- Use longest non-empty variant as canonical trading_name
            first_value(client_name) OVER (
                PARTITION BY tenant_id, lower(trim(client_name))
                ORDER BY length(client_name) DESC
            ),
            -- Use client_nuit if populated, else placeholder
            COALESCE(NULLIF(trim(client_nuit), ''), '000000000'),
            30,
            true,
            NOW(),
            NOW()
        FROM contracts
        WHERE client_name IS NOT NULL AND client_name != ''
        ON CONFLICT (tenant_id, nuit) DO NOTHING
    """))

    # Step 3: UPDATE contracts.client_id
    conn.execute(sa.text("""
        UPDATE contracts c
        SET client_id = cl.id
        FROM clients cl
        WHERE c.tenant_id = cl.tenant_id
          AND lower(trim(c.client_name)) = lower(trim(cl.trading_name))
          AND c.client_id IS NULL
    """))

    # Step 4: UPDATE billing_documents.client_id from their linked contract
    conn.execute(sa.text("""
        UPDATE billing_documents bd
        SET client_id = c.client_id,
            due_date = bd.issued_at + (c.payment_terms_days * INTERVAL '1 day')
        FROM contracts c
        WHERE bd.contract_id = c.id
          AND bd.client_id IS NULL
          AND c.client_id IS NOT NULL
    """))

    # Step 5: billing_documents without a contract — match by client_name directly
    conn.execute(sa.text("""
        UPDATE billing_documents bd
        SET client_id = cl.id
        FROM clients cl
        WHERE bd.tenant_id = cl.tenant_id
          AND lower(trim(bd.client_name)) = lower(trim(cl.trading_name))
          AND bd.client_id IS NULL
    """))

    conn.execute(sa.text("UPDATE billing_documents SET client_id = NULL, due_date = NULL"))
    conn.execute(sa.text("DELETE FROM clients WHERE nuit = '000000000'"))
