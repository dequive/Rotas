      const cookieStore = await cookies();
      const session = cookieStore.get("rotas_session")?.value;
      const month = request.nextUrl.searchParams.get("month") ?? "";

      const res = await fetch(

          },
        },
      );
      const data = await res.json();
      return NextResponse.json(data, { status: res.status });
    }

      async function triggerFuelReport() {
        setFuelState("pending");
        try {
          const res = await fetch(
            `/api/analytics/fuel-report?month=${currentMonth}`,
          );
          const data: JobStatus = await res.json();
          setFuelJobId(data.job_id);
          // Poll every 3s for up to 60s
          let attempts = 0;
          const interval = setInterval(async () => {
            attempts++;

              clearInterval(interval);
              setFuelState("failed");
              return;

            const pollData: JobStatus = await poll.json();
            if (pollData.status === "done") {
              clearInterval(interval);
              setFuelState("done");
            } else if (pollData.status === "failed") {
              clearInterval(interval);
              setFuelState("failed");

          setFuelState("failed");

    import { useEffect, useState } from "react";

    interface VehicleInsurance {
      id: string;
      policy_number: string;

    interface InsuranceFormData {
      policy_number: string;
      insurer: string;
      coverage_type: string;
      premium_amount: string;
      valid_from: string;
      valid_until: string;
      notes: string;
    }

    const EMPTY_FORM: InsuranceFormData = {
      policy_number: "",
      insurer: "",
      coverage_type: "civil_liability",
      premium_amount: "",
      valid_from: "",
      valid_until: "",
      notes: "",
    };

    const COVERAGE_LABELS: Record<string, string> = {
      civil_liability: "Responsabilidade Civil",
      comprehensive: "Multirriscos",
      cargo: "Cargo",
    };

    function insuranceStatus(validUntil: string): "vigente" | "a_renovar" | "expirado" {
      const today = new Date();
      const expiry = new Date(validUntil);
      const days = Math.floor((expiry.getTime() - today.getTime()) / 86400000);
      if (days < 0) return "expirado";
      if (days <= 30) return "a_renovar";
      return "vigente";
    }

    function StatusDot({ status }: { status: "vigente" | "a_renovar" | "expirado" }) {
      const map = {
        vigente: { dot: "bg-green-500", label: "Vigente", text: "text-green-700" },
        a_renovar: { dot: "bg-amber-400", label: "A renovar", text: "text-amber-700" },
        expirado: { dot: "bg-red-500", label: "Expirado", text: "text-red-700" },
      };
      const { dot, label, text } = map[status];
      return (
        <span className={`inline-flex items-center gap-1.5 text-xs font-semibold ${text}`}>
          <span className={`w-2 h-2 rounded-full ${dot}`} />
          {label}
        </span>
      );
    }

    export default function InsuranceTab({ vehicleId }: { vehicleId: string }) {
      const [insurances, setInsurances] = useState<VehicleInsurance[]>([]);
      const [loading, setLoading] = useState(true);
      const [showModal, setShowModal] = useState(false);
      const [form, setForm] = useState<InsuranceFormData>(EMPTY_FORM);
      const [submitting, setSubmitting] = useState(false);
      const [error, setError] = useState<string | null>(null);

      const API_BASE = process.env.NEXT_PUBLIC_ROTAS_API_BASE_URL ?? "";
      const token =
        typeof window !== "undefined" ? localStorage.getItem("rotas_access_token") : null;
      const tenantId =
        typeof window !== "undefined" ? localStorage.getItem("rotas_tenant_id") : null;

      function authHeaders() {
        return {
          "Content-Type": "application/json",
          ...(token ? { Authorization: `Bearer ${token}` } : {}),
          ...(tenantId ? { "X-Tenant-Id": tenantId } : {}),
        };
      }

      async function fetchInsurances() {
        setLoading(true);
        try {
          const res = await fetch(
            `${API_BASE}/api/v1/vehicles/${vehicleId}/insurance`,
            { headers: authHeaders() },
          );
          if (!res.ok) throw new Error("Failed to fetch");
          const data: VehicleInsurance[] = await res.json();
          setInsurances(data);
        } catch {
          setError("Não foi possível carregar as apólices.");
        } finally {
          setLoading(false);
        }
      }

      useEffect(() => {
        fetchInsurances();
        // eslint-disable-next-line react-hooks/exhaustive-deps
      }, [vehicleId]);

      async function handleSubmit(e: React.FormEvent) {
        e.preventDefault();
        setSubmitting(true);
        setError(null);
        try {
          const res = await fetch(
            `${API_BASE}/api/v1/vehicles/${vehicleId}/insurance`,
            {
              method: "POST",
              headers: authHeaders(),
              body: JSON.stringify({
                ...form,
                premium_amount: parseFloat(form.premium_amount),
              }),
            },
          );
          if (!res.ok) {
            const err = await res.json().catch(() => ({}));
            throw new Error((err as { detail?: string }).detail ?? "Erro ao registar");
          }
          setShowModal(false);
          setForm(EMPTY_FORM);
          await fetchInsurances();
        } catch (err) {
          setError(err instanceof Error ? err.message : "Erro desconhecido");
        } finally {

                        {ins.premium_amount.toLocaleString("pt-MZ", { minimumFractionDigits: 2 })}
                      </td>
                      <td className="px-4 py-3 font-mono text-sm">
                        {new Date(ins.valid_until).toLocaleDateString("pt-MZ")}
                      </td>
                      <td className="px-4 py-3">
                        <StatusDot status={insuranceStatus(ins.valid_until)} />
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          )}

          {/* Add Policy Modal */}
          {showModal && (
            <div className="fixed inset-0 bg-black/50 z-50 flex items-center justify-center p-4">
              <div className="bg-surface rounded-xl border border-line shadow-xl w-full max-w-lg">
                <div className="flex items-center justify-between px-6 py-4 border-b border-line">
                  <h3 className="text-lg font-bold text-ink">Registar Apólice</h3>
                  <button
                    onClick={() => { setShowModal(false); setError(null); }}
                    className="text-muted hover:text-ink text-xl leading-none"
                  >
                    ×
                  </button>
                </div>
                <form onSubmit={handleSubmit} className="px-6 py-4 space-y-4">
                  {error && <p className="text-red-600 text-sm">{error}</p>}
                  <div className="grid grid-cols-2 gap-4">
                    <div>
                      <label className="block text-xs font-semibold text-muted mb-1">
                        Número da Apólice
                      </label>
                      <input
                        required
                        value={form.policy_number}
                        onChange={(e) => setForm((f) => ({ ...f, policy_number: e.target.value }))}
                        className="w-full border border-line rounded-md px-3 py-2 text-sm bg-panel text-ink"
                      />
                    </div>
                    <div>
                      <label className="block text-xs font-semibold text-muted mb-1">Seguradora</label>
                      <input
                        required
                        value={form.insurer}
                        onChange={(e) => setForm((f) => ({ ...f, insurer: e.target.value }))}
                        className="w-full border border-line rounded-md px-3 py-2 text-sm bg-panel text-ink"
                      />
                    </div>

                    <div>

                        onChange={(e) => setForm((f) => ({ ...f, coverage_type: e.target.value }))}

                        <option value="comprehensive">Multirriscos</option>

                        required
                        type="number"
                        min="0"

                        value={form.premium_amount}
                        onChange={(e) => setForm((f) => ({ ...f, premium_amount: e.target.value }))}
                        className="w-full border border-line rounded-md px-3 py-2 text-sm bg-panel text-ink"

                    </div>
                  </div>
                  <div className="grid grid-cols-2 gap-4">
                    <div>
                      <label className="block text-xs font-semibold text-muted mb-1">
                        Válida De
                      </label>
                      <input

                        className="w-full border border-line rounded-md px-3 py-2 text-sm bg-panel text-ink"

                        Válida Até
                      </label>

                        value={form.valid_until}

                        className="w-full border border-line rounded-md px-3 py-2 text-sm bg-panel text-ink"

                    <label className="block text-xs font-semibold text-muted mb-1">Notas</label>

                    <button

                      onClick={() => { setShowModal(false); setError(null); }}
