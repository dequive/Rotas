## Goal

Build and test the pure-Python `check_driver_eligibility` function that reads existing Driver fields and returns an `EligibilityResult` with blocking reasons and expiring-soon warnings — no new Driver columns, no DB writes.

## Feature

**File:** `backend/app/modules/third_party/eligibility.py`

```python
from dataclasses import dataclass, field
from datetime import date
from app.modules.drivers.models import Driver

EXPIRY_WARNING_DAYS = 30


@dataclass
class EligibilityResult:
    is_eligible: bool
    blocking_reasons: list[str] = field(default_factory=list)
    expiring_soon: list[str] = field(default_factory=list)
    checked_at: date = field(default_factory=date.today)


def check_driver_eligibility(driver: Driver, reference_date: date) -> EligibilityResult:
    """Pure function — reads driver attributes, no DB access.

    Blocking conditions (is_eligible = False):
      - driver.status != 'active'
      - driver.license_valid_until < reference_date (expired)
      - driver.passport_valid_until < reference_date (expired)
      - driver.bi_valid_until < reference_date (expired)

    Warnings (is_eligible remains True, expiring_soon populated):
      - Any of the above dates is within EXPIRY_WARNING_DAYS of reference_date
    """
    blocking: list[str] = []
    expiring: list[str] = []

    if driver.status != "active":
        blocking.append(f"Driver status is '{driver.status}' — must be 'active'")

    _check_date(
        driver.license_valid_until, "Driving license", reference_date, blocking, expiring
    )
    _check_date(
        driver.passport_valid_until, "Passport", reference_date, blocking, expiring
    )
    _check_date(
        driver.bi_valid_until, "BI (national ID)", reference_date, blocking, expiring
    )

    return EligibilityResult(
        is_eligible=len(blocking) == 0,
        blocking_reasons=blocking,
        expiring_soon=expiring,
        checked_at=reference_date,
    )


def _check_date(
    expiry: date | None,
    label: str,
    reference_date: date,
    blocking: list[str],
    expiring: list[str],
) -> None:
    if expiry is None:
        return
    days_remaining = (expiry - reference_date).days
    if days_remaining < 0:
        blocking.append(f"{label} expired {abs(days_remaining)} day(s) ago")
    elif days_remaining <= EXPIRY_WARNING_DAYS:
        expiring.append(
            f"{label} expires in {days_remaining} day(s) on {expiry.isoformat()}"
        )
```

```python
"""Tests for check_driver_eligibility — pure function, no DB.

Test cases:
  1. test_suspended_driver_not_eligible — driver.status = 'suspended'
  2. test_expired_license_not_eligible — license_valid_until in the past
  3. test_expired_bi_not_eligible — bi_valid_until in the past
  4. test_all_clear_eligible — all dates in future, status active
  5. test_expiring_soon_warning — license expires in 15 days → warning, still eligible
"""
from datetime import date, timedelta
from unittest.mock import MagicMock
import pytest
from app.modules.third_party.eligibility import check_driver_eligibility, EligibilityResult

REF = date(2026, 6, 19)  # fixed reference date for deterministic tests


def make_driver(**overrides):
    """Create a mock Driver with all-valid defaults."""
    driver = MagicMock()
    driver.status = "active"
    driver.license_valid_until = REF + timedelta(days=365)
    driver.passport_valid_until = REF + timedelta(days=365)
    driver.bi_valid_until = REF + timedelta(days=365)
    for k, v in overrides.items():
        setattr(driver, k, v)
    return driver


def test_suspended_driver_not_eligible():
    driver = make_driver(status="suspended")
    result = check_driver_eligibility(driver, REF)
    assert result.is_eligible is False
    assert any("suspended" in r for r in result.blocking_reasons)


def test_expired_license_not_eligible():
    driver = make_driver(license_valid_until=REF - timedelta(days=1))
    result = check_driver_eligibility(driver, REF)
    assert result.is_eligible is False
    assert any("license" in r.lower() for r in result.blocking_reasons)


def test_expired_bi_not_eligible():
    driver = make_driver(bi_valid_until=REF - timedelta(days=10))
    result = check_driver_eligibility(driver, REF)
    assert result.is_eligible is False
    assert any("bi" in r.lower() for r in result.blocking_reasons)


def test_all_clear_eligible():
    driver = make_driver()
    result = check_driver_eligibility(driver, REF)
    assert result.is_eligible is True
    assert result.blocking_reasons == []
    assert result.expiring_soon == []


def test_expiring_soon_warning():
    driver = make_driver(license_valid_until=REF + timedelta(days=15))
    result = check_driver_eligibility(driver, REF)
    assert result.is_eligible is True      # warning, NOT blocking
    assert result.blocking_reasons == []
    assert any("15" in w for w in result.expiring_soon)
```
