```python
async def seed_work_order_with_task(client, headers, vehicle_id):
    """Create a maintenance request + work order in progress + one task. Returns (wo_id, task_id)."""
    # POST maintenance request (required before work order can be created)
    mr = await client.post("/api/v1/workshop/maintenance-requests", headers=headers, json={
        "vehicle_id": str(vehicle_id),
        "request_type": "corrective",
        "priority": "normal",
        "description": "Test maintenance",
    })
    assert mr.status_code == 200, mr.text
    mr_id = mr.json()["id"]

    # POST work order linked to the maintenance request
    wo = await client.post("/api/v1/workshop/work-orders", headers=headers, json={
        "maintenance_request_id": mr_id,
        "vehicle_id": str(vehicle_id),
        "planned_work": "Test work",
    })
    assert wo.status_code == 200, wo.text
    wo_id = wo.json()["id"]

    # Add a task to the work order
    task = await client.post(f"/api/v1/workshop/work-orders/{wo_id}/tasks", headers=headers, json={
        "description": "Test task",
    })
    assert task.status_code == 200, task.text
    task_id = task.json()["id"]

    return wo_id, task_id
```
