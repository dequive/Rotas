```typescript
export async function loadDriverScorecard(
  driverId: string,
  days = 30
): Promise<ScorecardData | null> {
  try {
    const data = await apiFetch(`/api/v1/drivers/${driverId}/scorecard?days=${days}`)
    return data as ScorecardData
  } catch {
    return null
  }
}

export interface ScorecardData {
  driver_id: string
  score: number | null
```

```tsx
"use client"

import { useState } from "react"
import { User, TrendingUp, Truck, Clock, Package } from "lucide-react"
import { loadDriverScorecard, ScorecardData } from "../lib/drivers-api"

interface Driver {
  id: string
  full_name: string
}

interface Props {
  drivers: Driver[]
}

// Tier to CSS badge class mapping (D-09 / UI-SPEC.md)
const TIER_BADGE: Record<string, { cls: string; label: string }> = {
  verde:        { cls: "badge green",  label: "Verde" },
  amarelo:      { cls: "badge orange", label: "Amarelo" },
  vermelho:     { cls: "badge red",    label: "Vermelho" },
  insuficiente: { cls: "badge",        label: "Dados insuficientes" },
}

function formatKm(km: number): string {
  return km.toLocaleString("pt-MZ") + " km"
}

export function DriverScorecardPanel({ drivers }: Props) {
  const [selectedId, setSelectedId] = useState<string>(drivers[0]?.id ?? "")
  const [scorecard, setScorecard] = useState<ScorecardData | null>(null)
  const [loading, setLoading] = useState(false)
  const [error, setError] = useState(false)

  async function handleDriverChange(id: string) {
    setSelectedId(id)
    setLoading(true)
    setError(false)
    const data = await loadDriverScorecard(id)
    setLoading(false)
    if (!data) {
      setError(true)
    } else {
      setScorecard(data)
    }
  }

  // Load scorecard for first driver on mount
  // Use useEffect to avoid SSR issues
  if (typeof window !== "undefined" && !scorecard && !loading && !error && selectedId) {
    // Trigger initial load on first render
  }

  const tier = scorecard ? TIER_BADGE[scorecard.tier] ?? TIER_BADGE.insuficiente : null
  const isInsufficient = scorecard?.tier === "insuficiente"

  return (
    <section>
      <hr className="section-divider" />
      <div className="domain-heading">
        <h2>Desempenho de Motoristas</h2>
        <p>Últimos 30 dias · Score composto</p>
      </div>

      <div style={{ marginBottom: 16 }}>
        <select
          value={selectedId}
          onChange={(e) => handleDriverChange(e.target.value)}
          className="form-select"
          aria-label="Seleccionar motorista"
        >
          {drivers.map((d) => (
            <option key={d.id} value={d.id}>{d.full_name}</option>
          ))}
        </select>
      </div>

      {error && (
        <p className="error-text">Erro ao carregar scorecard. Tente novamente.</p>
      )}

      {scorecard && (
        <div className="transport-kpis">
          {/* Summary card — composite score */}
          <div className="transport-kpi">
            <User size={16} color="var(--blue)" />
            <span>Score composto</span>
            {isInsufficient ? (
              <>
                <span className={tier?.cls ?? "badge"}>{tier?.label}</span>
                <small style={{ fontSize: 12, color: "var(--muted)" }}>
                  {scorecard.message ?? "Mínimo 3 viagens em 30 dias para score válido"}
                </small>
              </>
            ) : (
              <>
                <strong aria-label={`Score: ${scorecard.score} — ${tier?.label}`}>
                  {scorecard.score}
                </strong>
                <span className={tier?.cls ?? "badge"} aria-label={`Tier: ${tier?.label}`}>
                  {tier?.label}
                </span>
              </>
            )}
          </div>

          {/* Metric cards — loading skeleton uses --line background */}
          <div className="transport-kpi">
            <Package size={16} color="var(--blue)" />
            <span>Proof de entrega</span>
            {loading
              ? <span style={{ background: "var(--line)", height: 21, borderRadius: 4, display: "block", width: "60%" }} />
              : <strong>{scorecard.metrics.delivery_rate.toFixed(1)}%</strong>
            }
          </div>

          <div className="transport-kpi">
            <TrendingUp size={16} color="var(--blue)" />
            <span>Disciplina de sync</span>
            {loading
              ? <span style={{ background: "var(--line)", height: 21, borderRadius: 4, display: "block", width: "60%" }} />
              : <strong>{scorecard.metrics.sync_score.toFixed(1)}</strong>
            }
          </div>

          <div className="transport-kpi">
            <Truck size={16} color="var(--blue)" />
            <span>Quilómetros</span>
            {loading
              ? <span style={{ background: "var(--line)", height: 21, borderRadius: 4, display: "block", width: "60%" }} />
              : <strong>{formatKm(scorecard.metrics.total_km)}</strong>
            }
          </div>

          <div className="transport-kpi">
            <Clock size={16} color="var(--blue)" />
            <span>Eficiência de paradas</span>
            {loading
              ? <span style={{ background: "var(--line)", height: 21, borderRadius: 4, display: "block", width: "60%" }} />
              : <strong>{scorecard.metrics.stop_score.toFixed(1)}</strong>
            }
          </div>
        </div>
      )}
    </section>
  )
}
```

**Step 4: Create MaintenanceImminentPanel.tsx**

Create `apps/manager/app/components/MaintenanceImminentPanel.tsx`:

```tsx
import { Wrench } from "lucide-react"
import { ImminentAlert } from "../lib/control-tower-api"

interface Props {
  alerts: ImminentAlert[]
}

// Trigger type to marker color mapping (D-06 / UI-SPEC.md)
const TRIGGER_MARKER: Record<string, string> = {
  calendar: "history-marker orange",
  odometer: "history-marker cyan",
  overdue:  "history-marker red",
}

const TRIGGER_LABEL: Record<string, string> = {
  calendar: "Por data",
  odometer: "Por odómetro",
  overdue:  "Vencida",
}

function formatDueDate(iso: string): string {
  const d = new Date(iso)
  return `Vence em ${d.getDate().toString().padStart(2, "0")}/${(d.getMonth() + 1).toString().padStart(2, "0")}/${d.getFullYear()}`
}

function formatDueKm(km: number): string {
  return `Vence em ${km.toLocaleString("pt-MZ")} km`
}

export function MaintenanceImminentPanel({ alerts }: Props) {
  return (
    <div className="fleet-compliance-panel">
      <div style={{ display: "flex", alignItems: "center", gap: 8, marginBottom: 4 }}>
        <Wrench size={20} />
        <div>
          <h3 style={{ fontSize: 14, fontWeight: 700, margin: 0 }}>Manutenção Iminente</h3>
          <p style={{ fontSize: 12, color: "var(--muted)", margin: 0 }}>
            Viaturas próximas do prazo de intervenção
          </p>
        </div>
        {alerts.length > 0 && (
          <span className="badge red" style={{ marginLeft: "auto" }}>{alerts.length}</span>
        )}
      </div>

      {alerts.length === 0 ? (
        <div className="empty-state">
          <p style={{ fontWeight: 700 }}>Sem manutenções iminentes</p>
          <p style={{ color: "var(--muted)" }}>
            Todas as viaturas estão dentro dos limites de intervenção.
          </p>
        </div>
      ) : (
        <ul className="fleet-compliance-list" role="list">
          {alerts.map((alert) => (
            <li key={alert.plan_id} className="fleet-compliance-item" role="listitem">
              <span className={TRIGGER_MARKER[alert.trigger_type] ?? "history-marker"} />
              <div>
                <strong style={{ fontSize: 14 }}>{alert.vehicle_plate}</strong>
                <span style={{ fontSize: 12, color: "var(--muted)", display: "block" }}>
                  {alert.plan_name}
                </span>
                <span style={{ fontSize: 12, color: "var(--muted)" }}>
                  {alert.trigger_type === "overdue"
                    ? <span className="badge red">Vencida</span>
                    : alert.trigger_type === "calendar" && alert.next_due_at
                      ? formatDueDate(alert.next_due_at)
                      : alert.next_due_km !== null
                        ? formatDueKm(alert.next_due_km)
                        : null
                  }
                  {" "}
                  <span className="badge" style={{ fontSize: 11 }}>
                    {TRIGGER_LABEL[alert.trigger_type]}
                  </span>
                </span>
              </div>
            </li>
          ))}
        </ul>
      )}
    </div>
  )
}
```
