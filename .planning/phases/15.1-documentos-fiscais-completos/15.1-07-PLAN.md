class CartaPorteCreate(BaseModel):
    """Request body for POST /cargo/trips/{trip_id}/carta-porte (OPDOC-03).

    Carta de Porte Internacional — SADC CPI/CMR format for cross-border transport.
    """
    document_number: str         # CPI document number
    border_post: str             # border crossing point (required)
    country_destination: str     # destination country (required)
    sadc_cpi_number: str | None = None   # SADC-issued CPI number if available
    consignee_name: str | None = None    # receiver at destination
    consignee_nuit: str | None = None
    client_name: str | None = None       # shipper; defaults to contract client_name
    origin: str | None = None
    destination: str | None = None
    cargo_description: str | None = None
    gross_weight: float | None = None
    issued_at: datetime | None = None
    valid_from: datetime | None = None
    valid_until: datetime | None = None
    notes: str | None = None


class DavCreate(BaseModel):
    """Request body for POST /cargo/trips/{trip_id}/dav (OPDOC-04).

    DAV — Documento de Acompanhamento de Viagem issued by INATTER.
    This is a digital record of a physically issued document. No PDF generated.
    """
    document_number: str         # INATTER-issued number (required)
    authorization_code: str      # INATTER authorization code (required)
    inatter_office: str | None = None    # issuing INATTER office
    valid_routes: list[str] | None = None  # e.g. ["EN1", "EN6"]
    valid_from: datetime | None = None
    valid_until: datetime | None = None
    notes: str | None = None

def generate_carta_porte_pdf(
    document_number: str,
    client_name: str,           # shipper
    consignee_name: str | None,
    border_post: str,
    country_destination: str,
    sadc_cpi_number: str | None,
    origin: str | None,
    destination: str | None,
    cargo_description: str | None,
    gross_weight: float | None,
    issued_at: datetime | None,
    valid_from: datetime | None,
    valid_until: datetime | None,
    notes: str | None,
) -> ExportArtifact:
    """Generate A4 portrait PDF for Carta de Porte Internacional (OPDOC-03).

    Bilingual headers (PT/EN) as per SADC CPI format.
    Layout:
    - Header: ROTAS + CARTA DE PORTE INTERNACIONAL / INTERNATIONAL CONSIGNMENT NOTE
    - Document metadata block (number, date, SADC CPI number)
    - Shipper/Consignee two-column block
    - Border/destination block
    - Cargo description table
    - Customs declaration text block
    - Signature block (Shipper | Carrier | Customs)
    """
    pdf = FPDF(orientation="P", unit="mm", format="A4")  # portrait
    pdf.add_font("DejaVu", "", str(FONTS_DIR / "DejaVuSans.ttf"), uni=True)

    # Check for bold font variant
    bold_font = FONTS_DIR / "DejaVuSans-Bold.ttf"
    if bold_font.exists():
        pdf.add_font("DejaVu", "B", str(bold_font), uni=True)
    else:
        pdf.add_font("DejaVu", "B", str(FONTS_DIR / "DejaVuSans.ttf"), uni=True)

    pdf.set_auto_page_break(auto=True, margin=15)
    pdf.add_page()

    # ── Header ───────────────────────────────────────────────────────────────
    pdf.set_fill_color(*_NAV)
    pdf.rect(10, 10, 190, 20, style="F")
    pdf.set_font("DejaVu", "B", 12)
    pdf.set_text_color(*_WHITE)
    pdf.set_xy(12, 13)
    pdf.cell(0, 5, "CARTA DE PORTE INTERNACIONAL", new_x=XPos.LMARGIN, new_y=YPos.NEXT)
    pdf.set_font("DejaVu", "", 9)
    pdf.set_xy(12, 19)
    pdf.cell(0, 5, "INTERNATIONAL CONSIGNMENT NOTE (CMR/SADC)")

    # ── Document metadata ─────────────────────────────────────────────────────
    pdf.set_text_color(*_INK)
    pdf.set_xy(10, 36)
    pdf.set_font("DejaVu", "B", 8)
    pdf.cell(40, 5, "Número / Number:")
    pdf.set_font("DejaVu", "", 8)
    pdf.cell(60, 5, document_number)
    pdf.set_font("DejaVu", "B", 8)
    pdf.cell(40, 5, "Data / Date:")
    pdf.set_font("DejaVu", "", 8)
    pdf.cell(0, 5, _date(issued_at), new_x=XPos.LMARGIN, new_y=YPos.NEXT)

    if sadc_cpi_number:
        pdf.set_x(10)
        pdf.set_font("DejaVu", "B", 8)
        pdf.cell(40, 5, "Nº CPI SADC:")
        pdf.set_font("DejaVu", "", 8)
        pdf.cell(0, 5, sadc_cpi_number, new_x=XPos.LMARGIN, new_y=YPos.NEXT)

    # ── Shipper / Consignee ───────────────────────────────────────────────────
    pdf.ln(4)
    col_w = 90
    for label_pt, label_en, name in [
        ("Remetente", "Shipper", client_name or "—"),
        ("Destinatário", "Consignee", consignee_name or "—"),
    ]:
        x = pdf.get_x()
        y = pdf.get_y()
        pdf.set_fill_color(*_SOFT)
        pdf.rect(x, y, col_w, 24, style="F")
        pdf.set_font("DejaVu", "B", 8)
        pdf.set_xy(x + 2, y + 2)
        pdf.cell(col_w - 4, 4, f"{label_pt} / {label_en}")
        pdf.set_font("DejaVu", "", 9)
        pdf.set_xy(x + 2, y + 8)
        pdf.multi_cell(col_w - 4, 4, name, max_line_height=4)
        pdf.set_xy(x + col_w + 5, y)

    # ── Border / Destination ──────────────────────────────────────────────────
    pdf.ln(4)
    pdf.set_x(10)
    pdf.set_font("DejaVu", "B", 8)
    pdf.cell(50, 5, "Posto Fronteira / Border Post:")
    pdf.set_font("DejaVu", "", 9)
    pdf.cell(0, 5, border_post, new_x=XPos.LMARGIN, new_y=YPos.NEXT)
    pdf.set_x(10)
    pdf.set_font("DejaVu", "B", 8)
    pdf.cell(50, 5, "País de Destino / Country:")
    pdf.set_font("DejaVu", "", 9)
    pdf.cell(0, 5, country_destination, new_x=XPos.LMARGIN, new_y=YPos.NEXT)
    pdf.set_x(10)
    pdf.set_font("DejaVu", "B", 8)
    pdf.cell(30, 5, "Origem / Origin:")
    pdf.set_font("DejaVu", "", 8)
    pdf.cell(60, 5, origin or "—")
    pdf.set_font("DejaVu", "B", 8)
    pdf.cell(30, 5, "Destino / Destination:")
    pdf.set_font("DejaVu", "", 8)
    pdf.cell(0, 5, destination or "—", new_x=XPos.LMARGIN, new_y=YPos.NEXT)

    # ── Cargo table ───────────────────────────────────────────────────────────
    pdf.ln(4)
    pdf.set_fill_color(*_NAV)
    pdf.set_text_color(*_WHITE)
    pdf.set_font("DejaVu", "B", 8)
    pdf.set_x(10)
    pdf.cell(140, 6, "MERCADORIA / GOODS DESCRIPTION", fill=True)
    pdf.cell(50, 6, "PESO BRUTO / GROSS WEIGHT (kg)", align="C", fill=True, new_x=XPos.LMARGIN, new_y=YPos.NEXT)

    pdf.set_fill_color(*_SOFT)
    pdf.set_text_color(*_INK)
    pdf.set_font("DejaVu", "", 9)
    pdf.set_x(10)
    pdf.cell(140, 10, cargo_description or "—", fill=True)
    pdf.cell(50, 10, f"{gross_weight:.2f}" if gross_weight else "—", align="C", fill=True, new_x=XPos.LMARGIN, new_y=YPos.NEXT)

    # ── Customs declaration ───────────────────────────────────────────────────
    pdf.ln(6)
    pdf.set_x(10)
    pdf.set_font("DejaVu", "B", 7)
    pdf.cell(0, 4, "DECLARAÇÃO ADUANEIRA / CUSTOMS DECLARATION", new_x=XPos.LMARGIN, new_y=YPos.NEXT)
    pdf.set_font("DejaVu", "", 7)
    pdf.set_text_color(*_MUTED)
    pdf.set_x(10)
    pdf.multi_cell(190, 3.5,
        "O remetente confirma que as mercadorias acima descritas são correctas e que o transporte é efectuado nos "
        "termos e condições da Carta de Porte SADC. / The sender confirms that the goods described above are "
        "correct and that transport is carried out under SADC CMR terms and conditions."
    )

    if notes:
        pdf.ln(3)
        pdf.set_x(10)
        pdf.set_font("DejaVu", "B", 8)
        pdf.set_text_color(*_INK)
        pdf.cell(0, 5, "Observações / Notes:", new_x=XPos.LMARGIN, new_y=YPos.NEXT)
        pdf.set_font("DejaVu", "", 8)
        pdf.set_x(10)
        pdf.multi_cell(190, 4, notes)

    # ── Signature block ───────────────────────────────────────────────────────
    sig_y = max(pdf.get_y() + 10, 240)
    sig_w = 55
    for i, label in enumerate(["Remetente / Shipper", "Transportador / Carrier", "Alfândega / Customs"]):
        x = 10 + i * (sig_w + 7)
        pdf.line(x, sig_y, x + sig_w, sig_y)
        pdf.set_font("DejaVu", "", 7)
        pdf.set_text_color(*_MUTED)
        pdf.set_xy(x, sig_y + 1)
        pdf.cell(sig_w, 4, label, align="C")

    # ── Footer ────────────────────────────────────────────────────────────────
    pdf.set_xy(10, 283)
    pdf.set_font("DejaVu", "", 7)
    pdf.cell(0, 4, f"Gerado por ROTAS — {datetime.utcnow().strftime('%d/%m/%Y %H:%M')} UTC", align="C")

    buf = BytesIO()
    pdf.output(buf)
    content = buf.getvalue()
    safe_number = document_number.replace("/", "-")
    return ExportArtifact(
        filename=f"carta_porte_{safe_number}.pdf",
        content_type="application/pdf",
        content=content,
    )

async def create_carta_porte(
    db: AsyncSession,
    tenant_id: UUID,
    trip_id: UUID,
    payload: "CartaPorteCreate",
) -> dict:
    """OPDOC-03: Create Carta de Porte Internacional with bilingual PDF."""
    from app.modules.cargo.schemas import CartaPorteCreate
    from app.modules.cargo.exporters import generate_carta_porte_pdf

    trip = await _require_trip(db, tenant_id, trip_id)

    issued_at = payload.issued_at or now_utc()
    doc = TransportDocument(
        tenant_id=tenant_id,
        trip_id=trip_id,
        contract_id=trip.contract_id,
        document_type="carta_porte_internacional",
        document_number=payload.document_number,
        client_name=payload.client_name,
        origin=payload.origin or trip.origin,
        destination=payload.destination or trip.destination,
        issued_at=issued_at,
        valid_from=payload.valid_from,
        valid_until=payload.valid_until,
        status="pending",
        notes=payload.notes,
        extra_fields={
            "border_post": payload.border_post,
            "country_destination": payload.country_destination,
            "sadc_cpi_number": payload.sadc_cpi_number,
            "consignee_name": payload.consignee_name,
            "consignee_nuit": payload.consignee_nuit,
            "cargo_description": payload.cargo_description,
            "gross_weight": payload.gross_weight,
        },
    )
    db.add(doc)
    await db.flush()

    # Generate PDF
    artifact = generate_carta_porte_pdf(
        document_number=payload.document_number,
        client_name=payload.client_name or "—",
        consignee_name=payload.consignee_name,
        border_post=payload.border_post,
        country_destination=payload.country_destination,
        sadc_cpi_number=payload.sadc_cpi_number,
        origin=doc.origin,
        destination=doc.destination,
        cargo_description=payload.cargo_description,
        gross_weight=payload.gross_weight,
        issued_at=issued_at,
        valid_from=payload.valid_from,
        valid_until=payload.valid_until,
        notes=payload.notes,
    )
    stored_file = await save_generated_file(
        db,
        tenant_id,
        content=artifact.content,
        filename=artifact.filename,
        mime_type=artifact.content_type,
        file_type="pdf",
        entity_type="transport_document",
        entity_id=doc.id,
    )
    doc.file_id = stored_file.id

    await record_audit_log(
        db,
        tenant_id=tenant_id,
        action="cargo.transport_document.created",
        entity_type="transport_document",
        entity_id=doc.id,
        new_values={
            "document_type": "carta_porte_internacional",
            "document_number": payload.document_number,
            "border_post": payload.border_post,
            "country_destination": payload.country_destination,
            "trip_id": str(trip_id),
        },
    )
    await db.commit()
    await db.refresh(doc)

    result = serialize_transport_document(doc)
    result["pdf_url"] = f"/api/v1/cargo/transport-documents/{doc.id}/pdf"
    return result


async def create_dav(
    db: AsyncSession,
    tenant_id: UUID,
    trip_id: UUID,
    payload: "DavCreate",
) -> dict:
    """OPDOC-04: Create DAV/INATTER digital record. No PDF generated."""
    from app.modules.cargo.schemas import DavCreate

    trip = await _require_trip(db, tenant_id, trip_id)

    doc = TransportDocument(
        tenant_id=tenant_id,
        trip_id=trip_id,
        contract_id=trip.contract_id,
        document_type="dav",
        document_number=payload.document_number,
        valid_from=payload.valid_from,
        valid_until=payload.valid_until,
        status="pending",
        notes=payload.notes,
        extra_fields={
            "authorization_code": payload.authorization_code,
            "inatter_office": payload.inatter_office,
            "valid_routes": payload.valid_routes,
        },
    )
    db.add(doc)

    await record_audit_log(
        db,
        tenant_id=tenant_id,
        action="cargo.transport_document.created",
        entity_type="transport_document",
        entity_id=doc.id,
        new_values={
            "document_type": "dav",
            "document_number": payload.document_number,
            "authorization_code": payload.authorization_code,
            "trip_id": str(trip_id),
        },
    )
    await db.commit()
    await db.refresh(doc)
    return serialize_transport_document(doc)
