Playwright test patterns for Next.js App Router:
- `page.waitForLoadState('networkidle')` waits for all server-fetched data to arrive
- For Server Components, data is fetched server-side — no client-side network requests to intercept
- `page.waitForSelector('table, [data-empty-state], .empty-state, text="Sem dados"')` handles both loaded and empty states
- Timeout 10s for page loads in tests; 5s for element expectations (set in playwright.config.ts)

Existing CI structure (.github/workflows/ci.yml):
- `backend` job: postgres:16 + redis:7 services on ports 5432:5432 and 6379:6379
- `frontend` job: typecheck + next build (no backend needed)
- New `e2e` job must: depend on `backend`, have postgres:16 + redis:7, run migrations + seed, start backend + next dev, run playwright test
- The backend job already runs `psql ... -f infra/postgres-init.sql` (creates rotas_app role); e2e job must replicate this
- DATABASE_URL format in CI: `postgresql+asyncpg://rotas:rotas@localhost:5432/rotas`
- ALEMBIC_DATABASE_URL: `postgresql+psycopg://rotas:rotas@localhost:5432/rotas`
- ADMIN_DATABASE_URL: `postgresql+asyncpg://rotas_admin:rotas_admin_dev@localhost:5432/rotas`

wait-on package: `npx wait-on` is available via npx (part of npm). Use it to wait for both backend (8000) and Next.js (3030) to be ready before running Playwright.

```typescript
import { test, expect } from '@playwright/test';

// These tests do NOT use storageState — they test the login page itself.
// They override the storageState from playwright.config.ts by using test.use().
test.use({ storageState: { cookies: [], origins: [] } });

test.describe('Authentication', () => {
  test('login with valid credentials redirects to dashboard', async ({ page }) => {
    const email = process.env.E2E_EMAIL ?? 'e2e@rotas.local';
    const password = process.env.E2E_PASSWORD ?? 'E2eP@ss2024!';

    await page.goto('/login');
    await expect(page.locator('input[type="email"]')).toBeVisible();

    await page.locator('input[type="email"]').fill(email);
    await page.locator('input[type="password"]').fill(password);
    await page.getByRole('button', { name: 'Entrar' }).click();

    await page.waitForURL('/', { timeout: 15_000 });
    expect(page.url()).not.toContain('/login');
  });

  test('login with wrong password shows error message', async ({ page }) => {
    await page.goto('/login');
    await expect(page.locator('input[type="email"]')).toBeVisible();

    await page.locator('input[type="email"]').fill('e2e@rotas.local');
    await page.locator('input[type="password"]').fill('wrong-password-xyz');
    await page.getByRole('button', { name: 'Entrar' }).click();

    // Error paragraph appears: <p className="login-error">{error}</p>
    await expect(page.locator('p.login-error')).toBeVisible({ timeout: 8_000 });
    // URL stays on /login — no redirect happened
    expect(page.url()).toContain('/login');
  });

  test('unauthenticated visit to /viaturas redirects to /login', async ({ page }) => {
    await page.goto('/viaturas');
    // middleware.ts redirects unauthenticated requests to /login
    await page.waitForURL(/\/login/, { timeout: 8_000 });
    expect(page.url()).toContain('/login');
  });
});
```

**FILE: apps/manager/e2e/vehicles.spec.ts**

```typescript
import { test, expect } from '@playwright/test';

test.describe('Vehicles', () => {
  test('/viaturas list loads (table or empty state visible)', async ({ page }) => {
    await page.goto('/viaturas');
    await page.waitForLoadState('networkidle');

    // Page must render without a JS crash — either a table or an empty-state message
    // We wait for ANY of these: a table, a row, or an empty state indicator
    const tableOrEmpty = page.locator(
      'table, [role="table"], [data-empty-state], .empty-state, text=/Sem viaturas|Nenhuma viatura|Nenhum resultado/'
    );
    // If neither a table nor empty state appears within 10s, the page is broken
    await expect(tableOrEmpty.first()).toBeVisible({ timeout: 10_000 });
  });

  test('/viaturas/[id] detail page renders Seguros tab', async ({ page }) => {
    // Navigate to the list first to find a real vehicle ID, or if empty use a known test approach:
    // We navigate to /viaturas and look for the first row link.
    await page.goto('/viaturas');
    await page.waitForLoadState('networkidle');

    const firstRowLink = page.locator('table a, [role="row"] a, a[href^="/viaturas/"]').first();
    const firstLinkCount = await firstRowLink.count();

    if (firstLinkCount === 0) {
      // No vehicles seeded — skip detail test gracefully
      test.info().annotations.push({
        type: 'skip-reason',
        description: 'No vehicles in DB — skipping detail page test',
      });
      return;
    }

    await firstRowLink.click();
    await page.waitForLoadState('networkidle');

    // Vehicle detail page has tabs — "Seguros" tab must be present
    // Try role=tab first, then getByText as fallback
    const segurosTab = page.getByRole('tab', { name: /Seguros/i }).or(
      page.getByText(/Seguros/i, { exact: false })
    );
    await expect(segurosTab.first()).toBeVisible({ timeout: 10_000 });
  });
});
```
