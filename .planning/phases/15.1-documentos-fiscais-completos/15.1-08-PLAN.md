```python
_CHECKLIST_LABELS = {
    "cargo_manifest": "Manifesto de Carga",
    "load_permit": "Guia de Carregamento",
    "dav": "DAV / Guia INATTER",
    "guia_remessa": "Guia de Remessa",
    "carta_porte_internacional": "Carta de Porte Internacional (CPI)",
    "declaracao_carga_perigosa": "Declaração de Carga Perigosa",
}


async def get_document_checklist(
    db: AsyncSession,
    tenant_id: UUID,
    trip_id: UUID,
) -> dict:
    """OPDOC-05: Compute required vs present document checklist for a trip.

    Returns:
        {
            trip_id: UUID,
            is_international: bool,
            is_hazmat: bool,
            required: [
                {type, label, present, document_id}
            ]
        }
    """
    from sqlalchemy import select

    trip = await _require_trip(db, tenant_id, trip_id)

    # ── Determine trip characteristics ────────────────────────────────────────
    # is_international: use Trip.is_international if it exists, else check CPI presence
    is_international = getattr(trip, "is_international", None)
    if is_international is None:
        # Fallback: check if a carta_porte_internacional transport_document exists
        cpi_exists = await db.scalar(
            select(TransportDocument.id).where(
                TransportDocument.tenant_id == tenant_id,
                TransportDocument.trip_id == trip_id,
                TransportDocument.document_type == "carta_porte_internacional",
            ).limit(1)
        )
        is_international = cpi_exists is not None

    # is_hazmat: use Trip.is_hazmat if it exists, else check CargoManifest
    is_hazmat = getattr(trip, "is_hazmat", None)
    if is_hazmat is None:
        hazmat_manifest = await db.scalar(
            select(CargoManifest.id).where(
                CargoManifest.tenant_id == tenant_id,
                CargoManifest.trip_id == trip_id,
                CargoManifest.is_hazmat.is_(True),
            ).limit(1)
        )
        is_hazmat = hazmat_manifest is not None

    # ── Build required document type list ────────────────────────────────────
    required_types: list[str] = ["cargo_manifest", "load_permit"]
    if is_international:
        required_types.extend(["carta_porte_internacional", "guia_remessa"])
    else:
        required_types.extend(["dav", "guia_remessa"])
    if is_hazmat:
        if "declaracao_carga_perigosa" not in required_types:
            required_types.append("declaracao_carga_perigosa")

    # ── Query present documents ───────────────────────────────────────────────
    # Check cargo_manifest table
    manifest_row = await db.scalar(
        select(CargoManifest.id).where(
            CargoManifest.tenant_id == tenant_id,
            CargoManifest.trip_id == trip_id,
        ).order_by(CargoManifest.created_at.desc()).limit(1)
    )
    # Check load_permits table
    permit_row = await db.scalar(
        select(LoadPermit.id).where(
            LoadPermit.tenant_id == tenant_id,
            LoadPermit.trip_id == trip_id,
        ).order_by(LoadPermit.created_at.desc()).limit(1)
    )
    # Check transport_documents for remaining types
    transport_docs_result = await db.execute(
        select(TransportDocument.document_type, TransportDocument.id).where(
            TransportDocument.tenant_id == tenant_id,
            TransportDocument.trip_id == trip_id,
            TransportDocument.document_type.in_([
                "dav", "guia_remessa", "carta_porte_internacional", "declaracao_carga_perigosa"
            ]),
        )
    )
    transport_doc_map: dict[str, UUID] = {}
    for doc_type, doc_id in transport_docs_result:
        if doc_type not in transport_doc_map:  # keep first/latest
            transport_doc_map[doc_type] = doc_id

    # ── Build presence map ────────────────────────────────────────────────────
    presence: dict[str, tuple[bool, UUID | None]] = {
        "cargo_manifest": (manifest_row is not None, manifest_row),
        "load_permit": (permit_row is not None, permit_row),
    }
    for t in ["dav", "guia_remessa", "carta_porte_internacional", "declaracao_carga_perigosa"]:
        doc_id = transport_doc_map.get(t)
        presence[t] = (doc_id is not None, doc_id)

    # ── Assemble response ─────────────────────────────────────────────────────
    checklist = []
    for doc_type in required_types:
        present, doc_id = presence.get(doc_type, (False, None))
        checklist.append({
            "type": doc_type,
            "label": _CHECKLIST_LABELS.get(doc_type, doc_type),
            "present": present,
            "document_id": doc_id,
        })

    return {
        "trip_id": trip_id,
        "is_international": bool(is_international),
        "is_hazmat": bool(is_hazmat),
        "required": checklist,
    }
```
