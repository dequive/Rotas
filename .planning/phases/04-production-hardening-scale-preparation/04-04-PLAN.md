Metric weights:
- Delivery proof rate: 40% — delivered_trips / completed_trips in rolling 30d
- Sync discipline:     25% — sync_batches / trips ratio (floor at 1, cap at 3 per trip → normalize 0-100)

```python
async def get_driver_scorecard(
    db: AsyncSession,
    tenant_id: UUID,
    driver_id: UUID,
    days: int = 30,
) -> dict:
    """D-08: Compute driver scorecard from existing trip/sync/delivery data.

    Returns score 0-100 with tier, or None score if insufficient data (< 3 trips).
    All monetary/count aggregations use SQL — no Python-level iteration over rows.

    Formula (D-08 / RESEARCH.md Pattern 7):
    - 40% delivery proof rate (delivered / completed trips)
    - 25% sync discipline (avg sync batches per trip, normalized 0-100)
    - 20% distance (total km, normalized to fleet percentile — simplified: km/500 capped at 100)
    - 15% stop efficiency (avg stop min/km, inverted — 0 min/km = 100, ≥30 min/km = 0)
    """
    from app.modules.trips.models import Trip, TripStop
    from app.modules.cargo.models import DeliveryProof
    from app.modules.sync.models import SyncEvent

    window_start = now_utc() - timedelta(days=days)

    # --- Completed trips in window ---
    completed_count = await db.scalar(
        select(func.count(Trip.id)).where(
            Trip.tenant_id == tenant_id,
            Trip.driver_id == driver_id,
            Trip.status == "completed",
            Trip.actual_departure >= window_start,
        )
    ) or 0

    # Minimum viable window check (D-09 / Pitfall 5)
    if completed_count < 3:
        return {
            "driver_id": str(driver_id),
            "score": None,
            "tier": "insuficiente",
            "message": "Mínimo 3 viagens em 30 dias para score válido",
            "period_days": days,
            "completed_trips": completed_count,
            "metrics": {},
        }

    # --- Metric 1: Delivery proof rate (40%) ---
    # Count completed trips that have at least one delivery proof
    trips_with_proof = await db.scalar(
        select(func.count(func.distinct(Trip.id)))
        .join(DeliveryProof, DeliveryProof.trip_id == Trip.id)
        .where(
            Trip.tenant_id == tenant_id,
            Trip.driver_id == driver_id,
            Trip.status == "completed",
            Trip.actual_departure >= window_start,
        )
    ) or 0
    delivery_rate = (trips_with_proof / completed_count) * 100  # 0-100

    # --- Metric 2: Sync discipline (25%) ---
    # Count sync_events of type 'sync_batch' per trip, avg per trip, normalize 0-100
    # Assumption: sync_events table has driver_id + entity_type='trip' or count per trip_id
    # Use total sync events / completed_trips, floor 1, target ≥ 5 batches/trip = 100
    sync_count = await db.scalar(
        select(func.count(SyncEvent.id)).where(
            SyncEvent.tenant_id == tenant_id,
            SyncEvent.driver_id == driver_id,
            SyncEvent.created_at >= window_start,
        )
    ) or 0
    sync_per_trip = sync_count / completed_count
    # Normalize: 1 batch/trip = 20/100, 5+ batches/trip = 100/100
    sync_score = min(sync_per_trip / 5.0 * 100, 100.0)

    # --- Metric 3: Distance (20%) ---
    # Sum of (km_end - km_start) for completed trips, normalized: 500km/trip period = 100
    total_km_row = await db.execute(
        select(
            func.sum(Trip.km_end - Trip.km_start).label("total_km")
        ).where(
            Trip.tenant_id == tenant_id,
            Trip.driver_id == driver_id,
            Trip.status == "completed",
            Trip.actual_departure >= window_start,
            Trip.km_end.isnot(None),
            Trip.km_start.isnot(None),
        )
    )
    total_km = total_km_row.scalar() or 0
    # Normalize: 500 km over 30 days = 100 score
    distance_score = min((total_km / 500.0) * 100, 100.0)

    # --- Metric 4: Stop efficiency (15%) ---
    # Average stop duration in minutes per km of trip distance (inverted — lower is better)
    # Join TripStop to completed trips in window
    stop_data = await db.execute(
        select(
            func.avg(
                func.extract("epoch", TripStop.resumed_at - TripStop.stopped_at) / 60.0
            ).label("avg_stop_minutes")
        ).join(Trip, Trip.id == TripStop.trip_id)
        .where(
            Trip.tenant_id == tenant_id,
            Trip.driver_id == driver_id,
            Trip.status == "completed",
            Trip.actual_departure >= window_start,
            TripStop.resumed_at.isnot(None),
            TripStop.stopped_at.isnot(None),
        )
    )
    avg_stop_minutes = stop_data.scalar() or 0.0
    # Invert: 0 min avg = 100 score, 60+ min avg = 0 score
    stop_score = max(0.0, (1.0 - (avg_stop_minutes / 60.0)) * 100)

    # --- Composite score ---
    composite = (
        0.40 * delivery_rate
        + 0.25 * sync_score
        + 0.20 * distance_score
        + 0.15 * stop_score
    )
    score = round(composite, 1)

    # Tier assignment (D-09)
    if score >= 80:
        tier = "verde"
    elif score >= 60:
        tier = "amarelo"
    else:
        tier = "vermelho"

    return {
        "driver_id": str(driver_id),
        "score": score,
        "tier": tier,
        "period_days": days,
        "completed_trips": completed_count,
        "metrics": {
            "delivery_rate": round(delivery_rate, 1),
            "sync_score": round(sync_score, 1),
            "distance_score": round(distance_score, 1),
            "stop_score": round(stop_score, 1),
            "total_km": total_km,
            "avg_stop_minutes": round(avg_stop_minutes, 1),
        },
    }
```

```python
@router.get("/{driver_id}/scorecard")
async def get_driver_scorecard_endpoint(
    driver_id: UUID,
    days: int = Query(30, ge=7, le=90),
    db: AsyncSession = Depends(get_session),
    principal=Depends(get_current_principal),
):
    """D-11: Driver scorecard visible to manager roles only.

    Returns composite score 0-100 derived from rolling {days}-day window.
    Requires dashboard role (owner/admin/manager/viewer).
    Driver tokens (app='driver_app') are rejected with 403.
    """
    # Reject driver tokens — dashboard only (D-11)
    if getattr(principal, "app", None) == "driver_app":
        raise HTTPException(status_code=403, detail="Driver tokens cannot access scorecard")

    # Ensure driver belongs to this tenant
    from app.modules.drivers.service import _require_driver, get_driver_scorecard as _get_scorecard
    await _require_driver(db, principal.tenant_id, driver_id)

    return await _get_scorecard(db, principal.tenant_id, driver_id, days=days)
```
