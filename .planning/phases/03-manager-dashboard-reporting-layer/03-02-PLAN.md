**Step 2 — Fix N+1 in `_operational_close_queue`:**

The current loop does `await db.scalar(select(DeliveryProof)...)` and `await db.scalar(select(TripIncident.id)...)` per row. Replace with IN queries:

```python
async def _operational_close_queue(
    db: AsyncSession, tenant_id: UUID,
    *, page: int = 1, page_size: int = 50
) -> list[dict]:
    offset = (page - 1) * page_size
    # Base query — get trips
    rows = (await db.execute(
        select(Trip, Vehicle, Driver)
        .join(Vehicle, Trip.vehicle_id == Vehicle.id, isouter=True)
        .join(Driver, Trip.driver_id == Driver.id, isouter=True)
        .where(Trip.tenant_id == tenant_id, Trip.status == "closed")
        .limit(page_size).offset(offset)
    )).all()

    if not rows:
        return []

    trip_ids = [trip.id for trip, _, _ in rows]

    # Batch load delivery proofs — single IN query instead of N queries
    proofs_by_trip = {}
    proof_rows = (await db.execute(
        select(DeliveryProof)
        .where(
            DeliveryProof.tenant_id == tenant_id,
            DeliveryProof.trip_id.in_(trip_ids),
        )
        .order_by(DeliveryProof.delivered_at.desc(), DeliveryProof.created_at.desc())
    )).scalars().all()
    for proof in proof_rows:
        if proof.trip_id not in proofs_by_trip:
            proofs_by_trip[proof.trip_id] = proof

    # Batch load blocking incidents — single IN query
    incidents_by_trip = {}
    incident_rows = (await db.execute(
        select(TripIncident.id, TripIncident.trip_id)
        .where(
            TripIncident.tenant_id == tenant_id,
            TripIncident.trip_id.in_(trip_ids),
            TripIncident.resolution_status == "open",
        )
    )).all()
    for inc_id, trip_id in incident_rows:
        incidents_by_trip[trip_id] = inc_id

    # Assemble result (correlate in Python — O(N) dict lookup, no DB calls)
    result = []
    for trip, vehicle, driver in rows:
        proof = proofs_by_trip.get(trip.id)
        blocking_incident_id = incidents_by_trip.get(trip.id)
        result.append(_serialize_close_queue_item(trip, vehicle, driver, proof, blocking_incident_id))
    return result
```
