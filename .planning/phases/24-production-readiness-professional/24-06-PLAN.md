```python
"""Smoke tests for Phase 24 third-party extensions: contacts, ledger, evaluations."""
import uuid
from decimal import Decimal

import pytest
from httpx import AsyncClient


@pytest.mark.asyncio
async def test_contacts_crud(auth_client: AsyncClient, third_party_id: str):
    """Create, list, and delete a sub-contact."""
    # Create
    resp = await auth_client.post(
        f"/api/v1/third-party/{third_party_id}/contacts",
        json={"name": "Carlos Sitoe", "role": "Gerente", "phone": "+258820001234",
              "email": "carlos@example.com", "is_primary": True},
        headers={"Idempotency-Key": str(uuid.uuid4())},
    )
    assert resp.status_code == 201
    contact = resp.json()
    assert contact["name"] == "Carlos Sitoe"
    assert contact["is_primary"] is True
    contact_id = contact["id"]

    # List
    resp = await auth_client.get(f"/api/v1/third-party/{third_party_id}/contacts")
    assert resp.status_code == 200
    contacts = resp.json()
    assert any(c["id"] == contact_id for c in contacts)

    # Delete
    resp = await auth_client.delete(
        f"/api/v1/third-party/{third_party_id}/contacts/{contact_id}"
    )
    assert resp.status_code == 204

    # Verify deleted
    resp = await auth_client.get(f"/api/v1/third-party/{third_party_id}/contacts")
    assert not any(c["id"] == contact_id for c in resp.json())


@pytest.mark.asyncio
async def test_idempotency_on_contact_create(auth_client: AsyncClient, third_party_id: str):
    """Same Idempotency-Key returns cached response, not a duplicate."""
    idem_key = str(uuid.uuid4())
    payload = {"name": "Idem Test", "role": "Operador"}

    resp1 = await auth_client.post(
        f"/api/v1/third-party/{third_party_id}/contacts",
        json=payload,
        headers={"Idempotency-Key": idem_key},
    )
    assert resp1.status_code == 201
    id1 = resp1.json()["id"]

    resp2 = await auth_client.post(
        f"/api/v1/third-party/{third_party_id}/contacts",
        json=payload,
        headers={"Idempotency-Key": idem_key},
    )
    # Must return same id — not a new record
    assert resp2.json()["id"] == id1


@pytest.mark.asyncio
async def test_supplier_account_zero_balance(auth_client: AsyncClient, third_party_id: str):
    """New third party starts with zero balance."""
    resp = await auth_client.get(f"/api/v1/third-party/{third_party_id}/account")
    assert resp.status_code == 200
    data = resp.json()
    assert Decimal(data["balance"]) == Decimal("0.00")
    assert Decimal(data["total_debits"]) == Decimal("0.00")
    assert Decimal(data["total_credits"]) == Decimal("0.00")
    assert data["entries"] == []


@pytest.mark.asyncio
async def test_payment_creates_credit_entry(auth_client: AsyncClient, third_party_id: str):
    """POST /payments creates a credit entry and updates balance."""
    resp = await auth_client.post(
        f"/api/v1/third-party/{third_party_id}/payments",
        json={"amount": "5000.00", "description": "Pagamento teste", "payment_date": "2026-06-20"},
        headers={"Idempotency-Key": str(uuid.uuid4())},
    )
    assert resp.status_code == 201
    entry = resp.json()
    assert entry["entry_type"] == "credit"
    assert Decimal(entry["amount"]) == Decimal("5000.00")

    # Balance should now be 5000.00
    account_resp = await auth_client.get(f"/api/v1/third-party/{third_party_id}/account")
    account = account_resp.json()
    assert Decimal(account["balance"]) == Decimal("5000.00")
    assert len(account["entries"]) >= 1


@pytest.mark.asyncio
async def test_evaluation_weighted_score(auth_client: AsyncClient, third_party_id: str):
    """Evaluation score = weighted average of criteria."""
    criteria = [
        {"name": "prazo_entrega", "weight": 0.5, "score": 8.0},
        {"name": "qualidade", "weight": 0.3, "score": 7.0},
        {"name": "preco", "weight": 0.2, "score": 9.0},
    ]
    # Expected score: 0.5*8 + 0.3*7 + 0.2*9 = 4 + 2.1 + 1.8 = 7.9
    resp = await auth_client.post(
        f"/api/v1/third-party/{third_party_id}/evaluations",
        json={"criteria": criteria, "evaluation_date": "2026-06-20", "notes": "Bom fornecedor"},
        headers={"Idempotency-Key": str(uuid.uuid4())},
    )
    assert resp.status_code == 201
    ev = resp.json()
    assert abs(float(ev["score"]) - 7.9) < 0.01


@pytest.mark.asyncio
async def test_evaluation_invalid_weights_rejected(auth_client: AsyncClient, third_party_id: str):
    """Weights not summing to 1.0 must return 422."""
    criteria = [
        {"name": "prazo_entrega", "weight": 0.6, "score": 8.0},
        {"name": "qualidade", "weight": 0.6, "score": 7.0},  # 0.6 + 0.6 = 1.2 > 1.0
    ]
    resp = await auth_client.post(
        f"/api/v1/third-party/{third_party_id}/evaluations",
        json={"criteria": criteria},
        headers={"Idempotency-Key": str(uuid.uuid4())},
    )
    assert resp.status_code == 422


@pytest.mark.asyncio
async def test_provinces_seeded(anon_client: AsyncClient):
    """11 Mozambican provinces must be accessible via the provinces endpoint."""
    resp = await anon_client.get("/api/v1/third-party/provinces")
    assert resp.status_code == 200
    provinces = resp.json()
    assert len(provinces) == 11
```
