2. A Server Component (default export) that fetches data and renders the page

**Complete file:**

```tsx
import { DollarSign } from "lucide-react";
import Link from "next/link";
import { Suspense } from "react";
import { requireSession } from "@/app/lib/auth";
import { apiFetch } from "@/app/lib/api";
import { SidebarLayout } from "@/app/components/SidebarLayout";
import { PageHeader } from "@/app/components/ui/PageHeader";
import { SectionHeader } from "@/app/components/ui/SectionHeader";
import { KpiCard } from "@/app/components/ui/KpiCard";
import {
  DataTable,
  RotasTableHeader,
  RotasTableRow,
  RotasTableCell,
  TableBody,
  TableHeader,
  TableRow,
} from "@/app/components/ui/DataTable";

// ── Types ─────────────────────────────────────────────────────────────────────

interface ArSummary {
  current: string;
  "1_30": string;
  "31_60": string;
  "61_90": string;
  over_90: string;
  total_ar: string;
  currency: string;
  as_of: string;
}

interface TopDebtor {
  client_id: string;
  client_name: string;
  outstanding: string;
  worst_bucket: "current" | "1_30" | "31_60" | "61_90" | "over_90";
}

// ── Helpers ───────────────────────────────────────────────────────────────────

function formatMzn(val: string | number | null | undefined): string {
  if (val == null) return "—";
  const n = Number(val);
  return n.toLocaleString("pt-MZ", { minimumFractionDigits: 2, maximumFractionDigits: 2 });
}

// Bucket display config
const BUCKETS: Array<{
  key: keyof ArSummary;
  label: string;
  color: string;         // Tailwind class for the bucket card accent
  textColor: string;
}> = [
  { key: "current",  label: "Corrente",   color: "bg-slate-100 border-slate-200",   textColor: "text-slate-700" },
  { key: "1_30",     label: "1 – 30 d",   color: "bg-slate-100 border-slate-200",   textColor: "text-slate-700" },
  { key: "31_60",    label: "31 – 60 d",  color: "bg-amber-50 border-amber-300",    textColor: "text-amber-700" },
  { key: "61_90",    label: "61 – 90 d",  color: "bg-red-50 border-red-300",        textColor: "text-red-700"   },
  { key: "over_90",  label: "+ 90 d",     color: "bg-red-50 border-red-400",        textColor: "text-red-800"   },
];

const BUCKET_ORDER = ["current", "1_30", "31_60", "61_90", "over_90"];

function worstBucketLabel(bucket: string): string {
  const map: Record<string, string> = {
    current: "Corrente",
    "1_30": "1–30 d",
    "31_60": "31–60 d",
    "61_90": "61–90 d",
    over_90: "+90 d",
  };
  return map[bucket] ?? bucket;
}

function bucketBadgeClass(bucket: string): string {
  if (bucket === "over_90" || bucket === "61_90") return "text-red-700 bg-red-50 border border-red-200";
  if (bucket === "31_60") return "text-amber-700 bg-amber-50 border border-amber-200";
  return "text-slate-600 bg-slate-100 border border-slate-200";
}

// ── Page ─────────────────────────────────────────────────────────────────────

export default async function ArPage({
  searchParams,
}: {
  searchParams: Promise<{ as_of?: string }>;
}) {
  await requireSession();

  const sp = await searchParams;
  const asOf = sp.as_of ?? new Date().toISOString().slice(0, 10);
  const asOfParam = `?as_of=${asOf}`;

  let summary: ArSummary | null = null;
  let topDebtors: TopDebtor[] = [];

  try {
    summary = await apiFetch<ArSummary>(`/api/v1/billing/ar/summary${asOfParam}`);
  } catch {
    summary = null;
  }

  try {
    topDebtors = await apiFetch<TopDebtor[]>(`/api/v1/billing/ar/top-debtors${asOfParam}&limit=5`);
    if (!Array.isArray(topDebtors)) topDebtors = [];
  } catch {
    topDebtors = [];
  }

  // Compute total invoiced and total received for KPI cards.
  // The AR summary only has outstanding per bucket. We compute total_ar as total outstanding.
  // For total_invoiced and total_received we approximate from statement data or show "—"
  // if the top-debtors endpoint doesn't surface those. Use summary total_ar for "Total em Aberto".
  const totalAr = summary?.total_ar ?? "0";

  return (
    <SidebarLayout active="ar">
      <PageHeader
        title="Contas a Receber"
        description={`Referência: ${asOf}`}
      />

      {/* as_of date picker — simple form GET navigation */}
      <div className="mb-6 flex items-center gap-3">
        <label className="text-sm font-medium text-[var(--muted)]" htmlFor="as_of_input">
          Data de referência
        </label>
        <form method="GET" action="/ar" className="flex gap-2">
          <input
            id="as_of_input"
            type="date"
            name="as_of"
            defaultValue={asOf}
            className="px-3 py-1.5 text-sm border border-[var(--border)] rounded-md bg-white text-[var(--ink)] focus:outline-none focus:ring-1 focus:ring-[var(--amber)]"
          />
          <button
            type="submit"
            className="px-3 py-1.5 text-sm font-medium rounded-md bg-[var(--amber)] text-white hover:bg-[var(--amber-dark)] transition-colors"
          >
            Aplicar
          </button>
        </form>
      </div>

      {/* KPI strip — Total em Aberto as primary metric */}
      <div className="grid grid-cols-3 gap-4 mb-8">
        <KpiCard
          label="Total em Aberto"
          value={`MZN ${formatMzn(totalAr)}`}
          semantic={Number(totalAr) > 0 ? "error" : "default"}
        />
        <KpiCard
          label="Vencido (+30 d)"
          value={`MZN ${formatMzn(
            String(
              Number(summary?.["31_60"] ?? 0) +
              Number(summary?.["61_90"] ?? 0) +
              Number(summary?.over_90 ?? 0)
            )
          )}`}
          semantic={
            (Number(summary?.["61_90"] ?? 0) + Number(summary?.over_90 ?? 0)) > 0
              ? "error"
              : "warning"
          }
        />
        <KpiCard
          label="Clientes com Dívida"
          value={String(topDebtors.length)}
          semantic="default"
        />
      </div>

      {/* Aging grid — 5 bucket cards */}
      <SectionHeader title="Aging — Distribuição por Antiguidade" />
      <div className="grid grid-cols-5 gap-3 mb-8">
        {BUCKETS.map(({ key, label, color, textColor }) => {
          const amount = summary ? String(summary[key] ?? "0") : "0";
          return (
            <div
              key={key}
              className={`rounded-lg border p-4 ${color}`}
            >
              <div className={`text-xs font-semibold uppercase tracking-wide mb-2 ${textColor}`}>
                {label}
              </div>
              <div className={`font-mono text-lg font-medium ${textColor}`}>
                {formatMzn(amount)}
              </div>
              <div className="text-xs text-[var(--muted)] mt-1">MZN</div>
            </div>
          );
        })}
      </div>

      {/* Top debtors table */}
      <SectionHeader title="Maiores Devedores" />
      {topDebtors.length === 0 ? (
        <div className="text-sm text-[var(--muted)] py-6 text-center">
          Sem clientes com saldo em aberto.
        </div>
      ) : (
        <DataTable>
          <TableHeader>
            <TableRow>
              <RotasTableHeader>Cliente</RotasTableHeader>
              <RotasTableHeader align="right">Saldo em Aberto (MZN)</RotasTableHeader>
              <RotasTableHeader align="center">Pior Bucket</RotasTableHeader>
              <RotasTableHeader align="center">Detalhe</RotasTableHeader>
            </TableRow>
          </TableHeader>
          <TableBody>
            {topDebtors.map((debtor) => (
              <RotasTableRow key={debtor.client_id}>
                <RotasTableCell>
                  <span className="font-medium text-[var(--ink)]">{debtor.client_name}</span>
                </RotasTableCell>
                <RotasTableCell align="right">
                  <span className="font-mono text-sm font-medium text-[var(--ink)]">
                    {formatMzn(debtor.outstanding)}
                  </span>
                </RotasTableCell>
                <RotasTableCell align="center">
                  <span
                    className={`inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-xs font-medium ${bucketBadgeClass(debtor.worst_bucket)}`}
                  >
                    <span
                      className={`inline-block w-1.5 h-1.5 rounded-full ${
                        debtor.worst_bucket === "over_90" || debtor.worst_bucket === "61_90"
                          ? "bg-red-500"
                          : debtor.worst_bucket === "31_60"
                          ? "bg-amber-500"
                          : "bg-slate-400"
                      }`}
                    />
                    {worstBucketLabel(debtor.worst_bucket)}
                  </span>
                </RotasTableCell>
                <RotasTableCell align="center">
                  <Link
                    href={`/clientes/${debtor.client_id}`}
                    className="text-sm text-[var(--blue)] hover:underline"
                  >
                    Ver cliente
                  </Link>
                </RotasTableCell>
              </RotasTableRow>
            ))}
          </TableBody>
        </DataTable>
      )}
    </SidebarLayout>
  );
}
```

Key implementation notes:
- This is a pure Server Component (no `"use client"`). The date picker uses a standard HTML form
  GET to navigate to `/ar?as_of=YYYY-MM-DD` — no React state needed, no client-side JavaScript.
- `searchParams` is async in Next.js 14+ App Router — `await searchParams` before using.
- `apiFetch` is called server-side and injects auth cookies automatically.
- All monetary values use `font-mono` class (IBM Plex Mono per DESIGN.md).
- Amber for 31-60d bucket, red for 61-90d and over_90 — matches DESIGN.md requirement.
- Bucket badges use `::before` dot pattern via inline `<span>` (DESIGN.md: "Badges: always with
  a colored dot").
