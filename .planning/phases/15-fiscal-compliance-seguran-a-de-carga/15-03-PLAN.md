Add the `_assign_invoice_number()` helper function. Place it near the top of the service module, before `create_document()`:

```python
async def _assign_invoice_number(
    db: AsyncSession,
    document: "BillingDocument",
    tenant_id: UUID,
) -> str:
    """Assign a sequential invoice number to a BillingDocument at issue time.

    Sequence name: invoice_seq_{tenant_id_no_hyphens}_{year}
    Format: YYYY/NNNN (e.g., 2026/0001)

    Idempotent: if document.invoice_number is already set, returns it unchanged.
    Annual reset: one sequence per (tenant, year) — new sequence auto-created for each year.
    """
    if document.invoice_number:
        return document.invoice_number

    year = datetime.now(UTC).year
    seq_name = f"invoice_seq_{str(tenant_id).replace('-', '_')}_{year}"

    async def _next_val() -> str:
        result = await db.execute(text(f"SELECT nextval('{seq_name}')"))
        seq_val = result.scalar_one()
        return f"{year}/{seq_val:04d}"

    # Ensure sequence exists (lazy creation for new year or post-migration tenants)
    await db.execute(
        text(
            f"CREATE SEQUENCE IF NOT EXISTS \"{seq_name}\" "
            f"START 1 INCREMENT 1 NO MINVALUE NO MAXVALUE CACHE 1"
        )
    )

    invoice_number = await _next_val()
    document.invoice_number = invoice_number
    return invoice_number
```
