Part A — backend/app/core/rbac.py

Add at bottom of file, after require_platform_role():

    def require_own_tenant_or_platform(
        tenant_permission: str = ADMIN_USERS,
    ) -> Callable:
        """Combined guard for endpoints accessible to BOTH tenant admins and platform_admin.

        Resolves the principal from whichever scope the token belongs to:
          - scope="dashboard": delegates to require_permission(tenant_permission). Tenant user
            must hold the specified permission. Returns tenant Principal.
          - scope="platform": requires role=PLATFORM_ADMIN. Other platform roles (support,
            billing) are rejected with 403 — they use the dedicated /platform/* read endpoints
            instead. Returns platform Principal with tenant_id=None.
          - Any other scope: 401 from the underlying decoder.

        The returned principal MUST be inspected by the route handler:
          - If principal.scope == "platform": use path param {tenant_id} for tenant context.
          - If principal.scope == "dashboard": use principal.tenant_id (own tenant only).

        This guard does NOT inject the target tenant_id — routes remain responsible for
        choosing the correct tenant scope based on principal.scope.
        """
        from app.core.auth import (  # noqa: PLC0415
            Principal,
            get_current_platform_principal,
            get_current_principal,
        )

        async def dependency(
            authorization: Annotated[str | None, Header(alias="Authorization")] = None,
            x_tenant_id: Annotated[UUID | None, Header(alias="X-Tenant-Id")] = None,
        ) -> Principal:
            if not authorization or not authorization.lower().startswith("bearer "):
                raise ApiError(
                    "unauthorized",
                    "Authentication token is required.",
                    status_code=status.HTTP_401_UNAUTHORIZED,
                )
            token = authorization.split(" ", 1)[1]

            # Peek at scope claim without full validation to choose the right path.
            # Full validation happens in the delegated function — no security shortcut here.
            import base64  # noqa: PLC0415

            try:
                payload_b64 = token.split(".")[1]
                payload_b64 += "=" * (4 - len(payload_b64) % 4)
                import json  # noqa: PLC0415

                raw_claims = json.loads(base64.urlsafe_b64decode(payload_b64))
                token_scope = raw_claims.get("scope", "")
            except Exception:
                token_scope = ""

            if token_scope == "platform":
                # Full decode + DB lookup happens inside get_current_platform_principal.
                # Construct a minimal Request-like object since the dependency expects headers.
                from fastapi import Request  # noqa: PLC0415
                from fastapi.datastructures import Headers  # noqa: PLC0415

                # Call get_current_platform_principal directly with the Authorization value.
                principal = await get_current_platform_principal(authorization=authorization)
                # Platform path: only platform_admin allowed on tenant-plane endpoints.
                if principal.role != PLATFORM_ADMIN:
                    raise ApiError(
                        "forbidden",
                        "Only platform_admin can access tenant configuration endpoints directly."
                        " platform_support should use /platform/tenants/{id} read endpoints.",
                        status_code=status.HTTP_403_FORBIDDEN,
                    )
                return principal
            else:
                # Tenant path: full validation via get_current_principal.
                principal = await get_current_principal(
                    authorization=authorization, x_tenant_id=x_tenant_id
                )
                if not principal.has_any_permission(frozenset({tenant_permission})):
                    raise ApiError(
                        "forbidden",
                        "Insufficient permissions for this operation.",
                        status_code=status.HTTP_403_FORBIDDEN,
                        details={"required_permissions": [tenant_permission]},
                    )
                return principal

        return dependency

Part B — backend/app/modules/tenants/router.py

Replace every occurrence of `Depends(require_permission(ADMIN_USERS))` with
`Depends(require_own_tenant_or_platform())`.

Import require_own_tenant_or_platform from app.core.rbac. Remove require_permission import

Route handler adjustments for platform_admin path:
The four existing routes (GET /me, PATCH /me, GET /me/driver-despacho-table,
PUT /me/driver-despacho-table) currently call service with `principal.tenant_id`. When
principal.scope == "platform", principal.tenant_id is None — this would break.

Add a path param override pattern:

    @router.get("/me")
    async def get_my_tenant(
        principal: Annotated[Principal, Depends(require_own_tenant_or_platform())],
        db: Annotated[AsyncSession, Depends(get_session)],
        target_tenant_id: UUID | None = Query(None, alias="tenant_id"),
    ):
        if principal.scope == "platform":
            if target_tenant_id is None:
                raise ApiError(
                    "tenant_id_required",
                    "Platform admin must provide ?tenant_id= query param for this endpoint.",
                    status_code=status.HTTP_400_BAD_REQUEST,
                )
            effective_tenant_id = target_tenant_id
        else:
            effective_tenant_id = principal.tenant_id
        return await service.get_current_tenant(db, effective_tenant_id)

Apply the same pattern to PATCH /me, GET /me/driver-despacho-table,
PUT /me/driver-despacho-table.

For PATCH /me and PUT /me/driver-despacho-table, actor_id should be:
    principal.user_id if principal.scope != "platform" else None
(Platform admin mutations on tenant config are valid but have no user_id in the tenant plane;
pass actor_id=None — the platform_audit_logs already records the actor separately.)
