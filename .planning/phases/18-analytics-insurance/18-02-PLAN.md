    """ANA-02: Fuel consumption XLSX report generator."""
    from __future__ import annotations
    import calendar
    from uuid import UUID
    from sqlalchemy import func, select
    from sqlalchemy.ext.asyncio import AsyncSession
    from openpyxl import Workbook
    from openpyxl.styles import PatternFill, Font, Alignment
    from openpyxl.utils import get_column_letter
    from app.modules.fuel.models import FuelLog
    from app.modules.vehicles.models import Vehicle

    RED_FILL = PatternFill(start_color="FFCCCC", end_color="FFCCCC", fill_type="solid")
    HEADER_FONT = Font(bold=True)

    COLUMNS = [
        "Matrícula",
        "Combustível Total (L)",
        "Custo Total (MZN)",
        "Consumo Real (L/100km)",
        "Target (L/100km)",
        "Desvio %",
    ]

    async def generate_fuel_report_xlsx(
        db: AsyncSession, tenant_id: UUID, month: str
    ) -> bytes:
        """Generate XLSX with one row per vehicle for the given month (YYYY-MM).
        Red cell fill for Desvio % > 20.
        """
        year, mon = int(month[:4]), int(month[5:7])
        _, last_day = calendar.monthrange(year, mon)
        from datetime import date
        period_start = date(year, mon, 1)
        period_end = date(year, mon, last_day)

        # Query vehicles
        vehicles = (await db.execute(
            select(Vehicle).where(
                Vehicle.tenant_id == tenant_id,
                Vehicle.status == "active",
            ).limit(500)
        )).scalars().all()

        # Query fuel logs aggregated per vehicle
        fuel_rows = (await db.execute(
            select(
                FuelLog.vehicle_id,
                func.coalesce(func.sum(FuelLog.liters), 0).label("total_liters"),
                func.coalesce(func.sum(FuelLog.total_cost), 0).label("total_cost"),
            ).where(
                FuelLog.tenant_id == tenant_id,
                FuelLog.fuel_date >= period_start,
                FuelLog.fuel_date <= period_end,
            ).group_by(FuelLog.vehicle_id)
        )).all()

        fuel_map = {str(r.vehicle_id): r for r in fuel_rows}

        # Build trip km per vehicle for consumption calculation
        from app.modules.trips.models import Trip
        from datetime import datetime, timezone
        ps = datetime(year, mon, 1, tzinfo=timezone.utc)
        pe = datetime(year, mon, last_day, 23, 59, 59, tzinfo=timezone.utc)
        km_rows = (await db.execute(
            select(
                Trip.vehicle_id,
                func.coalesce(func.sum(Trip.km_end - Trip.km_start), 0).label("total_km"),
            ).where(
                Trip.tenant_id == tenant_id,
                Trip.status == "closed",
                Trip.closed_at >= ps,
                Trip.closed_at <= pe,
                Trip.km_end.isnot(None),
                Trip.km_start.isnot(None),
            ).group_by(Trip.vehicle_id)
        )).all()
        km_map = {str(r.vehicle_id): float(r.total_km) for r in km_rows}

        wb = Workbook()
        ws = wb.active
        ws.title = f"Combustível {month}"

        # Header
        for col_idx, col_name in enumerate(COLUMNS, start=1):
            cell = ws.cell(row=1, column=col_idx, value=col_name)
            cell.font = HEADER_FONT
            cell.alignment = Alignment(horizontal="center")

        # Data rows
        for row_idx, vehicle in enumerate(vehicles, start=2):
            vid = str(vehicle.id)
            fuel = fuel_map.get(vid)
            total_liters = float(fuel.total_liters) if fuel else 0.0
            total_cost = float(fuel.total_cost) if fuel else 0.0
            total_km = km_map.get(vid, 0.0)
            target = float(vehicle.avg_consumption_target) if vehicle.avg_consumption_target else None

            consumption = round((total_liters / total_km) * 100, 2) if total_km > 0 else None
            if consumption is not None and target is not None and target > 0:
                desvio = round(((consumption - target) / target) * 100, 1)
            else:
                desvio = None

            row_data = [
                vehicle.plate,
                round(total_liters, 2),
                round(total_cost, 2),
                consumption,
                target,
                desvio,
            ]
            for col_idx, value in enumerate(row_data, start=1):
                cell = ws.cell(row=row_idx, column=col_idx, value=value)
                if col_idx == 6 and desvio is not None and desvio > 20:
                    cell.fill = RED_FILL

        # Auto-width columns
        for col_idx in range(1, len(COLUMNS) + 1):
            ws.column_dimensions[get_column_letter(col_idx)].width = 20

        from io import BytesIO
        buf = BytesIO()
        wb.save(buf)
        return buf.getvalue()
    ```

    """ANA-03: Compliance expiry PDF report using fpdf2."""
    from __future__ import annotations
    from datetime import UTC, datetime, timedelta
    from pathlib import Path
    from uuid import UUID
    from sqlalchemy.ext.asyncio import AsyncSession
    from fpdf import FPDF

    # Locate DejaVuSans.ttf — same path used in billing exporters
    _FONT_PATH = Path(__file__).parent.parent / "assets" / "DejaVuSans.ttf"

    async def generate_compliance_report_pdf(
        db: AsyncSession, tenant_id: UUID
    ) -> bytes:
        """ANA-03: Compliance PDF with two sections: Vencidos and A vencer em 30 dias."""
        from app.modules.analytics.service import get_document_expiry_alerts
        from app.modules.tenants.models import Tenant
        from sqlalchemy import select

        tenant = await db.scalar(select(Tenant).where(Tenant.id == tenant_id))
        tenant_name = tenant.name if tenant else "ROTAS"

        all_alerts = await get_document_expiry_alerts(db, tenant_id, horizon_days=60)
        today = datetime.now(UTC).date()

        expired = [a for a in all_alerts if a["days_remaining"] < 0]
        expiring_30 = [a for a in all_alerts if 0 <= a["days_remaining"] <= 30]

        pdf = FPDF()
        pdf.set_auto_page_break(auto=True, margin=15)
        pdf.add_page()

        if _FONT_PATH.exists():
            pdf.add_font("DejaVu", "", str(_FONT_PATH), uni=True)
            pdf.set_font("DejaVu", size=10)
            title_font = ("DejaVu", 14)
            header_font = ("DejaVu", 10)
            body_font = ("DejaVu", 9)
        else:
            pdf.set_font("Helvetica", size=10)
            title_font = ("Helvetica", 14)
            header_font = ("Helvetica", 10)
            body_font = ("Helvetica", 9)

        # Header
        pdf.set_font(*title_font)
        pdf.cell(0, 10, f"Relatório de Conformidade — {tenant_name}", ln=True, align="C")
        pdf.set_font(*body_font)
        pdf.cell(0, 6, f"Gerado em: {today.strftime('%d/%m/%Y')}", ln=True, align="C")
        pdf.ln(6)

        def _write_section(title: str, items: list[dict]) -> None:
            pdf.set_font(*header_font)
            pdf.set_fill_color(240, 240, 240)
            pdf.cell(0, 8, title, ln=True, fill=True)
            pdf.set_font(*body_font)
            if not items:
                pdf.cell(0, 6, "  Nenhum documento nesta categoria.", ln=True)
                pdf.ln(2)
                return
            # Column headers
            pdf.set_font(*header_font)
            pdf.cell(40, 6, "Tipo", border=1)
            pdf.cell(60, 6, "Entidade", border=1)
            pdf.cell(40, 6, "Documento", border=1)
            pdf.cell(30, 6, "Vence em", border=1)
            pdf.cell(20, 6, "Dias", border=1, ln=True)
            pdf.set_font(*body_font)
            for item in items:
                pdf.cell(40, 6, item.get("entity_type", ""), border=1)
                pdf.cell(60, 6, str(item.get("entity_name", ""))[:28], border=1)
                pdf.cell(40, 6, str(item.get("document_type", ""))[:20], border=1)
                pdf.cell(30, 6, str(item.get("expires_at", ""))[:10], border=1)
                pdf.cell(20, 6, str(item.get("days_remaining", "")), border=1, ln=True)
            pdf.ln(4)

        _write_section("Documentos Vencidos", expired)
        _write_section("A vencer em 30 dias", expiring_30)

        return bytes(pdf.output())
    ```

    import uuid as _uuid

    @router.get("/fuel-report")
    async def trigger_fuel_report(
        request: Request,
        principal: Annotated[Principal, Depends(require_permission(FLEET_READ))],
        db: Annotated[AsyncSession, Depends(get_session)],
        month: Annotated[str, Query(description="Month YYYY-MM", pattern=r"^\d{4}-\d{2}$")],
    ) -> dict:
        """ANA-02: Enqueue fuel XLSX export job. Returns 202 with job_id.
        Idempotent: returns existing queued/processing job for same tenant+month.
        """
        from app.modules.billing.models import ExportJob
        from sqlalchemy import select

        existing = await db.scalar(
            select(ExportJob).where(
                ExportJob.tenant_id == principal.tenant_id,
                ExportJob.job_type == "analytics_fuel_xlsx",
                ExportJob.status.in_(["queued", "processing"]),
            )
        )
        if existing:
            from fastapi.responses import JSONResponse
            return JSONResponse(
                {"job_id": str(existing.id), "status": existing.status}, status_code=202
            )

        job = ExportJob(
            id=_uuid.uuid4(),
            tenant_id=principal.tenant_id,
            job_type="analytics_fuel_xlsx",
            status="queued",
        )
        db.add(job)
        await db.commit()
        await db.refresh(job)

        arq_redis = getattr(request.app.state, "arq_redis", None)
        if arq_redis:
            await arq_redis.enqueue_job(
                "task_export_fuel_report", str(job.id), str(principal.tenant_id), month
            )

        from fastapi.responses import JSONResponse
        return JSONResponse({"job_id": str(job.id), "status": "queued"}, status_code=202)

    async def task_export_fuel_report(ctx: dict, job_id: str, tenant_id: str, month: str) -> dict:
        """ANA-02: Generate fuel consumption XLSX and save via files module."""
        import structlog
        from uuid import UUID
        from sqlalchemy import select
        from app.modules.billing.models import ExportJob
        from app.modules.analytics.export_service import generate_fuel_report_xlsx
        from app.modules.files.service import save_generated_file

        logger = structlog.get_logger("worker")

        async with ctx["db_factory"]() as db:
            job = await db.scalar(select(ExportJob).where(ExportJob.id == UUID(job_id)))
            if not job:
                return {"error": "job_not_found"}
            job.status = "processing"
            await db.commit()

            try:
                content = await generate_fuel_report_xlsx(db, UUID(tenant_id), month)
                filename = f"fuel_report_{month}_{tenant_id[:8]}.xlsx"
                file_obj = await save_generated_file(
                    db,
                    UUID(tenant_id),
                    content=content,
                    filename=filename,
                    mime_type="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
                    file_type="analytics_fuel_xlsx",
                    entity_type="analytics",
                    entity_id=UUID(job_id),
                )
                job.status = "done"
                job.file_id = file_obj.id
                await db.commit()
                logger.info("task_export_fuel_report_done", job_id=job_id, month=month)
                return {"job_id": job_id, "status": "done", "file_id": str(file_obj.id)}
            except Exception as exc:
                job.status = "failed"
                job.error_message = str(exc)[:500]
                await db.commit()
                logger.error("task_export_fuel_report_failed", job_id=job_id, error=str(exc))
                return {"error": str(exc)}
