```python
def serialize_billing_document(document: BillingDocument, items: list[BillingItem]) -> dict:
    from datetime import date as _date

    # AR computed fields (FDOC-05)
    today = _date.today()
    days_overdue = 0
    aging_bucket = None
    if document.due_date and document.status == "issued":
        delta = (today - document.due_date).days
        days_overdue = max(0, delta)
        if days_overdue == 0:
            aging_bucket = "current"
        elif days_overdue <= 30:
            aging_bucket = "1_30"
        elif days_overdue <= 60:
            aging_bucket = "31_60"
        elif days_overdue <= 90:
            aging_bucket = "61_90"
        else:
            aging_bucket = "over_90"

    return {
        # ... all existing fields from the current implementation ...

        "parent_document_id": document.parent_document_id,
        "due_date": document.due_date,
        "client_nuit": document.client_nuit,
        "days_overdue": days_overdue,
        "aging_bucket": aging_bucket,
        "items": [serialize_billing_item(item) for item in items],
    }
```

```python
async def create_invoice_receipt(
    db: AsyncSession,
    tenant_id: UUID,
    parent_id: UUID,
    user_id: UUID,
) -> dict:
    """FDOC-04: Full payment — transition parent to paid and create Fatura-Recibo.

    Fatura-Recibo combines the invoice and proof of payment in one document.
    Parent must be in status 'issued'.
    """
    parent = await db.get(BillingDocument, parent_id)
    if not parent or parent.tenant_id != tenant_id:
        raise ApiError("billing_document_not_found", "Parent document not found.", status_code=404)
    if parent.status != "issued":
        raise ApiError(
            "invalid_parent_status",
            "Fatura-Recibo requires parent in status 'issued'.",
            status_code=409,
            details={"parent_status": parent.status},
        )

    paid_at = datetime.now(UTC)
    # Transition parent through SM-01
    await transition_billing_document(
        db,
        document=parent,
        new_status="paid",
        user_id=user_id,
        tenant_id=tenant_id,
        paid_at=paid_at,
    )

    doc = BillingDocument(
        tenant_id=tenant_id,
        contract_id=parent.contract_id,
        client_name=parent.client_name,
        contract_reference=parent.contract_reference,
        billing_period_start=parent.billing_period_start,
        billing_period_end=parent.billing_period_end,
        currency=parent.currency,
        subtotal=parent.subtotal,
        tax_amount=parent.tax_amount,
        total_amount=parent.total_amount,
        iva_rate=parent.iva_rate,
        status="paid",
        issued_at=paid_at,
        paid_at=paid_at,
        document_type="invoice_receipt",
        parent_document_id=parent_id,
        client_nuit=parent.client_nuit,
    )
    db.add(doc)
    await db.flush()
    await _assign_invoice_number(db, doc, tenant_id)
    await record_audit_log(
        db,
        tenant_id=tenant_id,
        action="billing.document_created",
        entity_type="billing_document",
        entity_id=doc.id,
        new_values={
            "document_type": "invoice_receipt",
            "parent_document_id": str(parent_id),
            "invoice_number": doc.invoice_number,
            "total_amount": str(doc.total_amount),
        },
    )
    await db.commit()
    await db.refresh(doc)
    result = serialize_billing_document(doc, [])
    result["parent_invoice_number"] = parent.invoice_number
    return result
```

```python
async def create_receipt(
    db: AsyncSession,
    tenant_id: UUID,
    parent_id: UUID,
    payload: "ReceiptCreate",
    user_id: UUID,
) -> dict:
    """FDOC-04: Standalone receipt — partial payment or payment confirmation.

    Does NOT change parent status. For full payment use create_invoice_receipt instead.
    """
    from app.modules.billing.schemas import ReceiptCreate

    parent = await db.get(BillingDocument, parent_id)
    if not parent or parent.tenant_id != tenant_id:
        raise ApiError("billing_document_not_found", "Parent document not found.", status_code=404)
    if parent.status not in ("issued", "paid", "overdue"):
        raise ApiError(
            "invalid_parent_status",
            "Receipt requires parent in status 'issued', 'paid', or 'overdue'.",
            status_code=409,
            details={"parent_status": parent.status},
        )

    issued_at = datetime.now(UTC)
    doc = BillingDocument(
        tenant_id=tenant_id,
        contract_id=parent.contract_id,
        client_name=parent.client_name,
        contract_reference=parent.contract_reference,
        billing_period_start=parent.billing_period_start,
        billing_period_end=parent.billing_period_end,
        currency=parent.currency,
        subtotal=payload.amount_paid,
        tax_amount=Decimal("0"),
        total_amount=payload.amount_paid,
        status="issued",
        issued_at=issued_at,
        document_type="receipt",
        parent_document_id=parent_id,
        client_nuit=payload.client_nuit or parent.client_nuit,
    )
    db.add(doc)
    await db.flush()
    await _assign_invoice_number(db, doc, tenant_id)
    await record_audit_log(
        db,
        tenant_id=tenant_id,
        action="billing.document_created",
        entity_type="billing_document",
        entity_id=doc.id,
        new_values={
            "document_type": "receipt",
            "parent_document_id": str(parent_id),
            "invoice_number": doc.invoice_number,
            "amount_paid": str(payload.amount_paid),
        },
    )
    await db.commit()
    await db.refresh(doc)
    result = serialize_billing_document(doc, [])
    result["parent_invoice_number"] = parent.invoice_number
    return result
```

**list_ar_documents:**

```python
async def list_ar_documents(
    db: AsyncSession,
    tenant_id: UUID,
    *,
    aging_bucket: str | None = None,
    contract_id: UUID | None = None,
    limit: int = 50,
    offset: int = 0,
) -> list[dict]:
    """FDOC-05: AR (Contas a Receber) — issued documents with due_date set.

    Returns documents ordered by days_overdue DESC.
    aging_bucket filter: current | 1_30 | 31_60 | 61_90 | over_90
    """
    from datetime import date, timedelta
    from sqlalchemy import and_, or_

    today = date.today()

    query = (
        select(BillingDocument)
        .where(
            BillingDocument.tenant_id == tenant_id,
            BillingDocument.status == "issued",
            BillingDocument.due_date.isnot(None),
        )
    )

    if contract_id:
        query = query.where(BillingDocument.contract_id == contract_id)

    # Apply aging_bucket filter in SQL for efficiency
    if aging_bucket == "current":
        query = query.where(BillingDocument.due_date >= today)
    elif aging_bucket == "1_30":
        query = query.where(
            BillingDocument.due_date < today,
            BillingDocument.due_date >= today - timedelta(days=30),
        )
    elif aging_bucket == "31_60":
        query = query.where(
            BillingDocument.due_date < today - timedelta(days=30),
            BillingDocument.due_date >= today - timedelta(days=60),
        )
    elif aging_bucket == "61_90":
        query = query.where(
            BillingDocument.due_date < today - timedelta(days=60),
            BillingDocument.due_date >= today - timedelta(days=90),
        )
    elif aging_bucket == "over_90":
        query = query.where(BillingDocument.due_date < today - timedelta(days=90))

    # Order by overdue days DESC (oldest overdue first) — due_date ASC = oldest first
    query = query.order_by(BillingDocument.due_date.asc()).limit(limit).offset(offset)
    rows = await db.execute(query)
    return [serialize_billing_document(doc, []) for doc in rows.scalars()]
```
