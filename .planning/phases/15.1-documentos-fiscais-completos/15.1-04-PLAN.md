```python
async def create_debit_note(
    db: AsyncSession,
    tenant_id: UUID,
    parent_id: UUID,
    payload: "DebitNoteCreate",
) -> dict:
    """FDOC-02: Create a Nota de Débito linked to an issued/paid invoice."""
    from app.modules.billing.schemas import DebitNoteCreate  # avoid circular at module level

    parent = await db.get(BillingDocument, parent_id)
    if not parent or parent.tenant_id != tenant_id:
        raise ApiError("billing_document_not_found", "Parent document not found.", status_code=404)
    if parent.status not in ("issued", "paid"):
        raise ApiError(
            "invalid_parent_status",
            "Nota de Débito requires parent document in status 'issued' or 'paid'.",
            status_code=409,
            details={"parent_status": parent.status},
        )

    from datetime import date, timedelta
    iva_amount = (payload.amount * payload.iva_rate).quantize(Decimal("0.01"))
    total = (payload.amount + iva_amount).quantize(Decimal("0.01"))
    due_date = payload.due_date or (date.today() + timedelta(days=30))

    doc = BillingDocument(
        tenant_id=tenant_id,
        contract_id=parent.contract_id,
        client_name=parent.client_name,
        contract_reference=parent.contract_reference,
        billing_period_start=parent.billing_period_start,
        billing_period_end=parent.billing_period_end,
        currency=parent.currency,
        subtotal=payload.amount,
        tax_amount=iva_amount,
        total_amount=total,
        iva_rate=payload.iva_rate,
        status="issued",
        issued_at=datetime.now(UTC),
        document_type="debit_note",
        parent_document_id=parent_id,
        due_date=due_date,
        client_nuit=payload.client_nuit or parent.client_nuit,
    )
    db.add(doc)
    await db.flush()
    await _assign_invoice_number(db, doc, tenant_id)
    await record_audit_log(
        db,
        tenant_id=tenant_id,
        action="billing.document_created",
        entity_type="billing_document",
        entity_id=doc.id,
        new_values={
            "document_type": "debit_note",
            "parent_document_id": str(parent_id),
            "invoice_number": doc.invoice_number,
            "amount": str(total),
            "reason": payload.reason,
        },
    )
    await db.commit()
    await db.refresh(doc)
    result = serialize_billing_document(doc, [])
    result["parent_invoice_number"] = parent.invoice_number
    result["reason"] = payload.reason
    return result
```
