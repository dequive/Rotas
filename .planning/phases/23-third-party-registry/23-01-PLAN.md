def _rls(table: str) -> None:
    """Apply v2.0 RLS rules to a tenant-scoped table."""
    op.execute(f"GRANT SELECT, INSERT, UPDATE, DELETE ON {table} TO rotas_app")
    op.execute(f"ALTER TABLE {table} ENABLE ROW LEVEL SECURITY")
    op.execute(f"ALTER TABLE {table} FORCE ROW LEVEL SECURITY")

        f"""CREATE POLICY tenant_isolation ON {table}

### Task 4 — Province seed script

**File:** `backend/scripts/seed_mz_provinces.py`

```python
"""Seed Mozambique provinces into mz_provinces table. Idempotent."""
import asyncio
from app.database import get_engine_for_scripts  # or create_async_engine directly

PROVINCES = [
    ("CD", "Cidade de Maputo", "Sul"),
    ("MP", "Maputo", "Sul"),
    ("GZ", "Gaza", "Sul"),
    ("IH", "Inhambane", "Sul"),
    ("SB", "Sofala", "Centro"),
    ("MN", "Manica", "Centro"),
    ("TE", "Tete", "Centro"),
    ("ZB", "Zambézia", "Norte"),
    ("NP", "Nampula", "Norte"),
    ("NI", "Niassa", "Norte"),
    ("CA", "Cabo Delgado", "Norte"),
]

async def seed() -> None:
    from sqlalchemy.ext.asyncio import create_async_engine
    from sqlalchemy import text
    from app.config import get_settings
    settings = get_settings()
    engine = create_async_engine(settings.database_url)
    async with engine.begin() as conn:
        for code, name, region in PROVINCES:
            await conn.execute(
                text(
                    "INSERT INTO mz_provinces (code, name, region) "
                    "VALUES (:code, :name, :region) "
                    "ON CONFLICT (code) DO NOTHING"
                ),
                {"code": code, "name": name, "region": region},
            )
    await engine.dispose()
    print(f"Seeded {len(PROVINCES)} provinces.")

if __name__ == "__main__":
    asyncio.run(seed())
```

Run with: `cd backend && python scripts/seed_mz_provinces.py`
