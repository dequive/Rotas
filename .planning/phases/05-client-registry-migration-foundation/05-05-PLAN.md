```python
import re

async def test_invoice_number_format(async_client, auth_headers, db, tenant_id):
    """CLI-05: Issued billing document has invoice_number in AAAA/NNNN format.

    _assign_invoice_number() is already implemented in billing/service.py (Phase 15).
    This test confirms the feature works end-to-end — no new backend code needed.
    """
    # 1. Create a contract (required by billing service)
    contract_res = await async_client.post(
        "/api/v1/contracts",
        json={
            "client_name": "Test Client CLI-05",
            "contract_reference": "TEST-CLI05",
            "billing_cycle": "monthly",
        },
        headers=auth_headers,
    )
    assert contract_res.status_code == 201, contract_res.text
    contract_id = contract_res.json()["id"]

    # 2. Create a billing document (draft)
    create_res = await async_client.post(
        "/api/v1/billing/documents",
        json={
            "contract_id": contract_id,
            "client_name": "Test Client CLI-05",
            "billing_period_start": "2026-01-01T00:00:00Z",
            "billing_period_end": "2026-01-31T23:59:59Z",
            "subtotal": "1000.00",
            "tax_amount": "0.00",
            "total_amount": "1000.00",
        },
        headers=auth_headers,
    )
    assert create_res.status_code == 201, create_res.text
    doc_id = create_res.json()["id"]

    # 3. Issue the document (triggers _assign_invoice_number)
    issue_res = await async_client.post(
        f"/api/v1/billing/documents/{doc_id}/issue",
        headers=auth_headers,
    )
    assert issue_res.status_code == 200, issue_res.text
    data = issue_res.json()

    # 4. Verify invoice_number format
    assert data.get("invoice_number") is not None, "invoice_number must be populated on issue"
    assert re.match(r'^\d{4}/\d{4}$', data["invoice_number"]), (
        f"invoice_number '{data['invoice_number']}' must match AAAA/NNNN format"
    )
```

Note: The exact endpoint paths for creating billing documents and issuing them may differ from above. Read the existing test_billing_api.py tests to find the correct endpoints and payload shape before writing this test.
