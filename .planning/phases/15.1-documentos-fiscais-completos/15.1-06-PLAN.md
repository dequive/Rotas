Create `backend/app/modules/cargo/exporters.py`. This module contains PDF generators for operational transport documents. Follow the same fpdf2 + DejaVuSans pattern used in `billing/exporters.py`.

```python
"""ROTAS cargo exporters — PDF generation for operational transport documents.

Uses fpdf2 + DejaVuSans for full UTF-8 support (Mozambican names with diacritics).
Font path: reuses billing/fonts/DejaVuSans.ttf (single source of truth).
"""
from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime
from io import BytesIO
from pathlib import Path

from fpdf import FPDF
from fpdf.enums import XPos, YPos

# Reuse font files from billing module — same package, no duplication
FONTS_DIR = Path(__file__).parent.parent / "billing" / "fonts"

# Brand palette (matches billing/exporters.py)
_NAV = (16, 32, 51)
_SOFT = (245, 247, 250)
_LINE = (216, 222, 232)
_INK = (23, 32, 51)
_MUTED = (102, 112, 133)
_WHITE = (255, 255, 255)


@dataclass(frozen=True)
class ExportArtifact:
    filename: str
    content_type: str
    content: bytes


def _date(value: datetime | None, fmt: str = "%d/%m/%Y") -> str:
    if not isinstance(value, datetime):
        return "—"
    return value.strftime(fmt)


def generate_guia_remessa_pdf(
    document_number: str,
    client_name: str,       # shipper
    recipient_name: str,
    recipient_nuit: str | None,
    origin: str | None,
    destination: str | None,
    cargo_description: str | None,
    package_count: int | None,
    gross_weight: float | None,
    issued_at: datetime | None,
    notes: str | None,
) -> ExportArtifact:
    """Generate A4 landscape PDF for Guia de Remessa (OPDOC-02).

    Layout:
    - Header: ROTAS logo area + document title + document number
    - Two-column section: Remetente (shipper) | Destinatário (recipient)
    - Cargo table: Descrição | Volumes | Peso Bruto (kg)
    - Signature block: Remetente | Transportador | Destinatário
    - Footer: generated timestamp
    """
    pdf = FPDF(orientation="L", unit="mm", format="A4")  # landscape
    pdf.add_font("DejaVu", "", str(FONTS_DIR / "DejaVuSans.ttf"), uni=True)
    pdf.add_font("DejaVu", "B", str(FONTS_DIR / "DejaVuSans-Bold.ttf"), uni=True)
    pdf.set_auto_page_break(auto=True, margin=15)
    pdf.add_page()

    # ── Header ───────────────────────────────────────────────────────────────
    pdf.set_fill_color(*_NAV)
    pdf.rect(10, 10, 277, 18, style="F")
    pdf.set_font("DejaVu", "B", 14)
    pdf.set_text_color(*_WHITE)
    pdf.set_xy(12, 13)
    pdf.cell(0, 8, "GUIA DE REMESSA", new_x=XPos.LMARGIN, new_y=YPos.NEXT)

    pdf.set_font("DejaVu", "", 9)
    pdf.set_xy(180, 13)
    pdf.cell(107, 4, f"Nº: {document_number}", align="R", new_x=XPos.LMARGIN, new_y=YPos.NEXT)
    pdf.set_xy(180, 17)
    pdf.cell(107, 4, f"Data: {_date(issued_at)}", align="R")

    # ── Shipper / Recipient columns ───────────────────────────────────────────
    pdf.set_text_color(*_INK)
    pdf.set_xy(10, 34)

    col_w = 133  # half of 277 - 10 margins
    for label, name, nuit in [
        ("REMETENTE (Shipper)", client_name or "—", None),
        ("DESTINATÁRIO (Recipient)", recipient_name, recipient_nuit),
    ]:
        x = pdf.get_x()
        y = pdf.get_y()
        pdf.set_fill_color(*_SOFT)
        pdf.rect(x, y, col_w, 28, style="F")
        pdf.set_font("DejaVu", "B", 8)
        pdf.set_xy(x + 2, y + 2)
        pdf.cell(col_w - 4, 5, label)
        pdf.set_font("DejaVu", "", 10)
        pdf.set_xy(x + 2, y + 8)
        pdf.multi_cell(col_w - 4, 5, name, max_line_height=5)
        if nuit:
            pdf.set_font("DejaVu", "", 8)
            pdf.set_xy(x + 2, y + 20)
            pdf.cell(col_w - 4, 4, f"NUIT: {nuit}")
        pdf.set_xy(x + col_w + 4, y)  # move to right column

    # ── Route bar ────────────────────────────────────────────────────────────
    pdf.ln(4)
    pdf.set_x(10)
    pdf.set_font("DejaVu", "B", 8)
    pdf.cell(20, 5, "Origem:")
    pdf.set_font("DejaVu", "", 8)
    pdf.cell(100, 5, origin or "—")
    pdf.set_font("DejaVu", "B", 8)
    pdf.cell(20, 5, "Destino:")
    pdf.set_font("DejaVu", "", 8)
    pdf.cell(0, 5, destination or "—", new_x=XPos.LMARGIN, new_y=YPos.NEXT)

    # ── Cargo table ──────────────────────────────────────────────────────────
    pdf.ln(4)
    pdf.set_fill_color(*_NAV)
    pdf.set_text_color(*_WHITE)
    pdf.set_font("DejaVu", "B", 8)
    pdf.set_x(10)
    pdf.cell(157, 6, "DESCRIÇÃO DA MERCADORIA", fill=True)
    pdf.cell(30, 6, "VOLUMES", align="C", fill=True)
    pdf.cell(40, 6, "PESO BRUTO (kg)", align="C", fill=True, new_x=XPos.LMARGIN, new_y=YPos.NEXT)

    pdf.set_fill_color(*_SOFT)
    pdf.set_text_color(*_INK)
    pdf.set_font("DejaVu", "", 9)
    pdf.set_x(10)
    pdf.cell(157, 10, cargo_description or "—", fill=True)
    pdf.cell(30, 10, str(package_count) if package_count else "—", align="C", fill=True)
    pdf.cell(40, 10, f"{gross_weight:.2f}" if gross_weight else "—", align="C", fill=True, new_x=XPos.LMARGIN, new_y=YPos.NEXT)

    # ── Notes ────────────────────────────────────────────────────────────────
    if notes:
        pdf.ln(3)
        pdf.set_x(10)
        pdf.set_font("DejaVu", "B", 8)
        pdf.cell(0, 5, "Observações:", new_x=XPos.LMARGIN, new_y=YPos.NEXT)
        pdf.set_font("DejaVu", "", 8)
        pdf.set_x(10)
        pdf.multi_cell(277, 4, notes)

    # ── Signature block ───────────────────────────────────────────────────────
    sig_y = max(pdf.get_y() + 10, 170)
    sig_w = 80
    for i, label in enumerate(["Remetente", "Transportador", "Destinatário"]):
        x = 10 + i * (sig_w + 8)
        pdf.line(x, sig_y, x + sig_w, sig_y)
        pdf.set_font("DejaVu", "", 7)
        pdf.set_text_color(*_MUTED)
        pdf.set_xy(x, sig_y + 1)
        pdf.cell(sig_w, 4, label, align="C")

    # ── Footer ────────────────────────────────────────────────────────────────
    pdf.set_text_color(*_MUTED)
    pdf.set_font("DejaVu", "", 7)
    pdf.set_xy(10, 195)
    pdf.cell(0, 4, f"Gerado por ROTAS — {datetime.utcnow().strftime('%d/%m/%Y %H:%M')} UTC", align="C")

    buf = BytesIO()
    pdf.output(buf)
    content = buf.getvalue()
    safe_number = document_number.replace("/", "-")
    return ExportArtifact(
        filename=f"guia_remessa_{safe_number}.pdf",
        content_type="application/pdf",
        content=content,
    )
```

```python
from app.modules.files.service import save_generated_file  # add to imports
from app.modules.cargo.exporters import generate_guia_remessa_pdf  # add to imports


async def create_guia_remessa(
    db: AsyncSession,
    tenant_id: UUID,
    trip_id: UUID,
    payload: "GuiaRemessaCreate",
) -> dict:
    """OPDOC-02: Create Guia de Remessa transport document with PDF generation."""
    from app.modules.cargo.schemas import GuiaRemessaCreate

    trip = await _require_trip(db, tenant_id, trip_id)

    issued_at = payload.issued_at or now_utc()
    doc = TransportDocument(
        tenant_id=tenant_id,
        trip_id=trip_id,
        contract_id=trip.contract_id,
        document_type="guia_remessa",
        document_number=payload.document_number,
        client_name=payload.client_name,
        recipient_name=payload.recipient_name,
        recipient_nuit=payload.recipient_nuit,
        origin=payload.origin or trip.origin,
        destination=payload.destination or trip.destination,
        issued_at=issued_at,
        status="pending",
        notes=payload.notes,
        extra_fields={
            "cargo_description": payload.cargo_description,
            "package_count": payload.package_count,
            "gross_weight": payload.gross_weight,
        },
    )
    db.add(doc)
    await db.flush()

    # Generate PDF
    artifact = generate_guia_remessa_pdf(
        document_number=payload.document_number,
        client_name=payload.client_name or "—",
        recipient_name=payload.recipient_name,
        recipient_nuit=payload.recipient_nuit,
        origin=doc.origin,
        destination=doc.destination,
        cargo_description=payload.cargo_description,
        package_count=payload.package_count,
        gross_weight=payload.gross_weight,
        issued_at=issued_at,
        notes=payload.notes,
    )
    stored_file = await save_generated_file(
        db,
        tenant_id,
        content=artifact.content,
        filename=artifact.filename,
        mime_type=artifact.content_type,
        file_type="pdf",
        entity_type="transport_document",
        entity_id=doc.id,
    )
    doc.file_id = stored_file.id

    await record_audit_log(
        db,
        tenant_id=tenant_id,
        action="cargo.transport_document.created",
        entity_type="transport_document",
        entity_id=doc.id,
        new_values={
            "document_type": "guia_remessa",
            "document_number": payload.document_number,
            "trip_id": str(trip_id),
            "recipient_name": payload.recipient_name,
        },
    )
    await db.commit()
    await db.refresh(doc)

    result = serialize_transport_document(doc)
    result["pdf_url"] = f"/api/v1/cargo/transport-documents/{doc.id}/pdf"
    return result
```
