```python
"""Phase 7 AR correctness tests — bucket assignment, outstanding calculation, PDF generation."""
from datetime import UTC, date, datetime, timedelta
from decimal import Decimal

import pytest
from sqlalchemy.ext.asyncio import AsyncSession

from app.modules.billing.models import BillingDocument, ClientPayment, PaymentAllocation
from app.modules.billing.service import (
    generate_client_statement_pdf,
    get_ar_summary,
    get_client_statement,
)
from app.modules.clients.models import Client
from app.modules.tenants.models import Tenant


# ── Helpers ───────────────────────────────────────────────────────────────────


def _make_invoice(
    tenant_id,
    client_id,
    total_amount: Decimal,
    due_days_ago: int,
    issued_days_ago: int = 60,
    status: str = "issued",
    document_type: str = "invoice",
) -> BillingDocument:
    now = datetime.now(UTC)
    return BillingDocument(
        tenant_id=tenant_id,
        client_id=client_id,
        client_name="Test Client",
        contract_reference="REF-001",
        billing_period_start=now - timedelta(days=90),
        billing_period_end=now - timedelta(days=60),
        currency="MZN",
        subtotal=total_amount,
        tax_amount=Decimal("0.00"),
        total_amount=total_amount,
        status=status,
        document_type=document_type,
        due_date=now - timedelta(days=due_days_ago),
        issued_at=now - timedelta(days=issued_days_ago),
        issuer_name="Test Issuer",
        iva_rate=Decimal("0.16"),
    )


async def _make_client(db: AsyncSession, tenant_id, name: str = "Test Client") -> Client:
    c = Client(tenant_id=tenant_id, trading_name=name, is_active=True)
    db.add(c)
    await db.flush()
    return c


# ── Tests ─────────────────────────────────────────────────────────────────────


@pytest.mark.asyncio
async def test_ar_summary_buckets_correct(db: AsyncSession, tenant: Tenant):
    """Invoice with due_date 45 days ago must appear in 31_60 bucket."""
    client = await _make_client(db, tenant.id)
    invoice = _make_invoice(tenant.id, client.id, Decimal("1000.00"), due_days_ago=45)
    db.add(invoice)
    await db.commit()

    result = await get_ar_summary(db, tenant.id)

    assert result["31_60"] == Decimal("1000.00"), f"Expected 1000 in 31_60, got {result}"
    assert result["current"] == Decimal("0.00")
    assert result["1_30"] == Decimal("0.00")
    assert result["61_90"] == Decimal("0.00")
    assert result["over_90"] == Decimal("0.00")
    assert result["total_ar"] == Decimal("1000.00")
    assert "as_of" in result


@pytest.mark.asyncio
async def test_ar_summary_as_of_param(db: AsyncSession, tenant: Tenant):
    """Invoice issued today must NOT appear when as_of=yesterday."""
    client = await _make_client(db, tenant.id)
    today = datetime.now(UTC)
    invoice = BillingDocument(
        tenant_id=tenant.id,
        client_id=client.id,
        client_name="Test Client",
        contract_reference="REF-002",
        billing_period_start=today - timedelta(days=30),
        billing_period_end=today,
        currency="MZN",
        subtotal=Decimal("500.00"),
        tax_amount=Decimal("0.00"),
        total_amount=Decimal("500.00"),
        status="issued",
        document_type="invoice",
        due_date=today + timedelta(days=30),  # not yet due — current bucket
        issued_at=today,  # issued today
        issuer_name="Test Issuer",
        iva_rate=Decimal("0.16"),
    )
    db.add(invoice)
    await db.commit()

    # Query as of yesterday — invoice issued today should be excluded
    yesterday = (today - timedelta(days=1)).date()
    result = await get_ar_summary(db, tenant.id, as_of=yesterday)

    assert result["total_ar"] == Decimal("0.00"), (
        f"Invoice issued today should be excluded when as_of=yesterday, got {result}"
    )


@pytest.mark.asyncio
async def test_ar_summary_excludes_drafts(db: AsyncSession, tenant: Tenant):
    """Draft documents must not appear in AR summary regardless of due_date."""
    client = await _make_client(db, tenant.id)
    draft_invoice = _make_invoice(
        tenant.id, client.id, Decimal("2000.00"), due_days_ago=10, status="draft"
    )
    db.add(draft_invoice)
    await db.commit()

    result = await get_ar_summary(db, tenant.id)

    assert result["total_ar"] == Decimal("0.00"), (
        f"Draft invoice must not appear in AR, got {result}"
    )
    assert result["current"] == Decimal("0.00")
    assert result["1_30"] == Decimal("0.00")


@pytest.mark.asyncio
async def test_client_statement_outstanding_correct(db: AsyncSession, tenant: Tenant, owner):
    """Invoice 1000 MZN, payment 400 MZN confirmed+allocated → outstanding == 600.00."""
    client = await _make_client(db, tenant.id)
    invoice = _make_invoice(tenant.id, client.id, Decimal("1000.00"), due_days_ago=15)
    db.add(invoice)
    await db.flush()

    payment = ClientPayment(
        tenant_id=tenant.id,
        client_id=client.id,
        amount=Decimal("400.00"),
        currency="MZN",
        value_date=datetime.now(UTC),
        payment_method="transfer",
        status="confirmed",
        created_by=owner.id,
    )
    db.add(payment)
    await db.flush()

    allocation = PaymentAllocation(
        tenant_id=tenant.id,
        payment_id=payment.id,
        billing_document_id=invoice.id,
        amount_applied=Decimal("400.00"),
    )
    db.add(allocation)
    await db.commit()

    result = await get_client_statement(db, tenant.id, client.id)

    assert result["total_invoiced"] == Decimal("1000.00")
    assert result["total_paid"] == Decimal("400.00")
    assert result["balance"] == Decimal("600.00")

    # Per-document outstanding must be present and correct
    docs = result["documents"]
    assert len(docs) >= 1
    doc = next(d for d in docs if d.get("id") == invoice.id or str(d.get("id")) == str(invoice.id))
    assert Decimal(str(doc["outstanding"])) == Decimal("600.00"), (
        f"Per-document outstanding must be 600.00, got {doc.get('outstanding')}"
    )
    assert Decimal(str(doc["amount_paid"])) == Decimal("400.00"), (
        f"Per-document amount_paid must be 400.00, got {doc.get('amount_paid')}"
    )


@pytest.mark.asyncio
async def test_client_statement_pdf_returns_bytes(db: AsyncSession, tenant: Tenant, owner):
    """generate_client_statement_pdf returns bytes with PDF magic bytes, len > 1000."""
    client = await _make_client(db, tenant.id, "Cliente Moçambicano Ção")
    invoice = _make_invoice(tenant.id, client.id, Decimal("750.00"), due_days_ago=20)
    db.add(invoice)
    await db.commit()

    pdf_bytes = await generate_client_statement_pdf(db, tenant.id, client.id)

    assert isinstance(pdf_bytes, bytes), "generate_client_statement_pdf must return bytes"
    assert len(pdf_bytes) > 1000, f"PDF too small: {len(pdf_bytes)} bytes"
    assert pdf_bytes[:4] == b"%PDF", "Response is not a PDF (missing %PDF magic bytes)"


@pytest.mark.asyncio
async def test_ar_cross_tenant(db: AsyncSession, tenant: Tenant):
    """Tenant A invoices must not appear in tenant B AR summary."""
    # Create tenant B
    tenant_b = Tenant(name="Tenant B", slug="tenant-b", plan="starter")
    db.add(tenant_b)
    await db.flush()

    client_a = await _make_client(db, tenant.id, "Client A")
    invoice_a = _make_invoice(tenant.id, client_a.id, Decimal("3000.00"), due_days_ago=50)
    db.add(invoice_a)
    await db.commit()

    # Query tenant B — should see nothing
    result_b = await get_ar_summary(db, tenant_b.id)

    assert result_b["total_ar"] == Decimal("0.00"), (
        f"Tenant B must not see Tenant A invoices, got {result_b}"
    )
```
