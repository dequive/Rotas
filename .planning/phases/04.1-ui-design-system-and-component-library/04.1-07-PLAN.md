    const Icon = meta.icon;
    const statusTrips = trips.filter((trip) => trip.status === status);
    return (
      <WorkQueue
        key={status}
        icon={Icon}
        title={meta.label}
        tone={meta.tone as "blue" | "orange" | "red" | "green" | "cyan" | "amber"}

        items={statusTrips.map((trip) => ({

          reference: trip.plate,
          title: trip.route,
          meta: money(trip.amount),
        }))}
      />
    );
  })}
